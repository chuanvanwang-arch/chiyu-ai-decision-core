"""Action 表面注册表（R1–R6 基数规则的**可执行形式**）—— ★ 新建，demo 未收敛。

⛔ 为什么要有注册表而不是继续加端点
──────────────────────────────────
demo 的 59 个端点里出现"同一能力多端点并存"（如 `/api/retro/*` 4 个），
这是 R5（一能力一 Action）被违反的典型。加端点不会报错，所以没人拦得住 ——
除非有一个注册表在**注册期**就拒绝重复。

R1–R6（判定规则）：
    R1 一个业务域 5–9 个 Action（过多=CRUD 爆炸，过少=能力不足）
    R2 Action 名 = 动词 + 对象（`submit_request` 而非 `requestSubmit`）
    R3 读操作不单独占位（读并入对象的 query Action）
    R4 写操作必须有决策锚（decision_id）
    R5 一个能力一个 Action（同能力的多入口必须折叠）
    R6 高危写必须有 force 确认位

⛔ 零领域词：注册表只承载**判定机制**；具体 Action 清单是 L4（消费方域内容）。
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple

ACTION_MIN, ACTION_MAX = 5, 9
_NAME_RE = re.compile(r"^[a-z][a-z0-9]*(_[a-z0-9]+)+$")
_HIGH_RISK_HINTS = ("delete", "purge", "reset", "drop", "approve", "release")


@dataclass(frozen=True)
class ActionSpec:
    name: str
    capability: str                 # R5：同一 capability 只能有一个 Action
    mutating: bool = False          # R4：写操作必须有决策锚
    high_risk: bool = False         # R6：高危写必须有 force
    has_force: bool = False

    def violations(self) -> List[str]:
        out: List[str] = []
        if not _NAME_RE.match(self.name):
            out.append(f"R2 命名不合规（须 verb_object）：{self.name}")
        if self.high_risk and not self.has_force:
            out.append(f"R6 高危写缺 force 确认位：{self.name}")
        if not self.high_risk and self.has_force:
            out.append(f"R6 非高危动作不应带 force 位：{self.name}")
        return out


class ActionRegistry:
    def __init__(self) -> None:
        self._by_name: Dict[str, ActionSpec] = {}
        self._by_capability: Dict[str, str] = {}

    def register(self, spec: ActionSpec) -> None:
        if spec.name in self._by_name:
            raise ValueError(f"重复注册 Action：{spec.name}")
        owner = self._by_capability.get(spec.capability)
        if owner is not None:
            raise ValueError(
                f"R5 违反：capability {spec.capability!r} 已由 {owner!r} 占用，"
                f"新增 {spec.name!r} ⇒ 应折叠到既有 Action，而不是新开端点")
        self._by_name[spec.name] = spec
        self._by_capability[spec.capability] = spec.name

    def audit(self) -> Tuple[List[str], List[str]]:
        """返回 (errors, warnings)。errors 非空 ⇒ 表面不合规，不得发布。"""
        errors: List[str] = []
        warnings: List[str] = []
        n = len(self._by_name)
        if n < ACTION_MIN:
            warnings.append(f"R1 Action 数偏少（{n} < {ACTION_MIN}）")
        if n > ACTION_MAX:
            errors.append(f"R1 Action 数超限（{n} > {ACTION_MAX}）⇒ CRUD 爆炸")
        for spec in self._by_name.values():
            errors.extend(spec.violations())
        return errors, warnings

    def by_name(self, name: str) -> Optional[ActionSpec]:
        return self._by_name.get(name)

    def names(self) -> Tuple[str, ...]:
        return tuple(sorted(self._by_name))


__all__ = ["ActionSpec", "ActionRegistry", "ACTION_MIN", "ACTION_MAX"]

"""决策脊柱审批闸（WS3 + B3 三态）—— 硬约束先于置信度（A 红线优先 / 反馈闭环金律26）。

⛔ 与 `chiyu_kernel.interaction.ApprovalGate` 是**两个不同层**，职责不重叠、可组合：
   - 本模块 `DecisionApprovalGate`：决策**落库前**的审批裁定（红线/置信度 ⇒ 自动放行 or 转人工）。
   - `interaction.ApprovalGate`：工具**执行前**的人工确认（高危险工具 HITL）。
   命名加 `Decision` 前缀，避免门面与 interaction 同名符号静默覆盖（同 `vector_*` 前缀纪律）。
   组合方式：本闸 `requires_human=True` ⇒ 消费方再走 `interaction.ApprovalGate` 做真人确认。

⛔ B3 三态（方案 Z，2026-10-09）：裁定从两态升三态。
   此前 `requires_human=True`（等人批）与 `redline`（硬拒）共用 `approved=False`，
   脊柱只能一律记 `approval_rejected` —— 「待人工审批」被记成「已驳回」（错值），
   且消费方无法从结果区分"该去催人"还是"该改方案"。
   兼容性：旧消费方读 `.approved` 仍工作（PENDING/REJECTED 下均为 False）；
   `DecisionPassThroughGate` 默认仍 APPROVED（未注入闸的既有消费方零影响）；
   **红线命中恒 REJECTED，不因 pending 豁免**（A 红线优先）。
"""
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
from typing import Optional, Protocol, Tuple


class DecisionVerdictState(str, Enum):
    """审批裁定三态（B3 方案 Z）。

    - `APPROVED`：可自动落库（脊柱走 persister）；
    - `REJECTED`：硬拒（红线命中），**不可**转 pending、不可自动豁免；
    - `PENDING`：待人工审批（判据缺失 / 未达自主阈值），不落正向边、审计留 `approval_pending`。
    """

    APPROVED = "approved"
    REJECTED = "rejected"
    PENDING = "pending"


@dataclass(frozen=True)
class DecisionApprovalVerdict:
    """审批裁定（三态）。``approved`` / ``pending`` 为派生属性（旧消费方读 `.approved` 不破）。"""

    state: DecisionVerdictState = DecisionVerdictState.APPROVED
    reason: str = ""
    requires_human: bool = False

    @property
    def approved(self) -> bool:
        """兼容字段：仅 APPROVED 为真（旧两态消费方读此属性不破）。"""
        return self.state is DecisionVerdictState.APPROVED

    @property
    def pending(self) -> bool:
        """兼容字段：PENDING 为真（供消费方区分"待人工"与"已驳回"）。"""
        return self.state is DecisionVerdictState.PENDING

    @property
    def rejected(self) -> bool:
        """兼容字段：REJECTED 为真。"""
        return self.state is DecisionVerdictState.REJECTED


class DecisionApprovalGate(Protocol):
    """决策脊柱 S4→S5 审批闸协议（鸭子类型，消费方可注入自有实现）。"""

    def decide(self, *, decision_id: str, confidence: float, redline_hit: bool,
               redline_reasons: Tuple[str, ...], autonomous: bool,
               rubric_missing: Tuple[str, ...], elements_missing: Tuple[str, ...],
               threshold: float) -> DecisionApprovalVerdict:
        ...


@dataclass
class RedlinePriorityGate:
    """默认审批闸 —— embody A 红线优先 + 硬约束先于置信度（金律26）。

    裁定顺序（不可重排，顺序即语义）：
      1. 红线命中 → **REJECTED**（不因置信度高而豁免，也不得降级为 pending）。
      2. 判据缺失（九尺子/八要素）→ **PENDING**（转人工确认，不自动放行）。
      3. ``autonomous`` 为真 ⇒ APPROVED；否则 PENDING（未达自主阈值，等人批）。
    """

    def decide(self, *, decision_id: str, confidence: float, redline_hit: bool,
               redline_reasons: Tuple[str, ...], autonomous: bool,
               rubric_missing: Tuple[str, ...], elements_missing: Tuple[str, ...],
               threshold: float) -> DecisionApprovalVerdict:
        # 1. 硬约束先于置信度：红线命中立即硬拒，高置信度不豁免（A 红线优先 / 金律26）
        if redline_hit:
            reason = "红线命中（硬约束优先于置信度）"
            if redline_reasons:
                reason += "：" + "；".join(redline_reasons)
            return DecisionApprovalVerdict(
                state=DecisionVerdictState.REJECTED, reason=reason, requires_human=True)
        # 2. 判据缺失 → 待人工确认（B3：等人批 ≠ 已驳回）
        missing = tuple(list(rubric_missing) + list(elements_missing))
        if missing:
            return DecisionApprovalVerdict(
                state=DecisionVerdictState.PENDING,
                reason="判据缺失需人工确认：" + "、".join(missing),
                requires_human=True)
        # 3. autonomous 为真（红线未命中且置信度达门槛）⇒ 自动放行；否则待人工
        if autonomous:
            return DecisionApprovalVerdict(
                state=DecisionVerdictState.APPROVED, reason="置信度达标，自动放行")
        return DecisionApprovalVerdict(
            state=DecisionVerdictState.PENDING,
            reason=f"未达自主放行条件（置信度 {confidence:.3f}），需人工确认",
            requires_human=True)


@dataclass
class DecisionPassThroughGate:
    """审批闸透传（默认）：不拦截，等价于未注入审批闸（保留 run_spine 既有行为）。"""

    def decide(self, *, decision_id: str, confidence: float, redline_hit: bool,
               redline_reasons: Tuple[str, ...], autonomous: bool,
               rubric_missing: Tuple[str, ...], elements_missing: Tuple[str, ...],
               threshold: float) -> DecisionApprovalVerdict:
        return DecisionApprovalVerdict(
            state=DecisionVerdictState.APPROVED, reason="无审批闸（透传）")

"""SHA-256 链式审计（SKILL #10 的机制层）—— ★ 内核新建，demo 无此目录。

⛔ 为什么是链式而不是逐条哈希
────────────────────────────
逐条哈希只能证明"这条没被改"；链式还能证明"**没有中间记录被删掉**"——
后者才是审计被绕过的常见形式（删掉不利记录，其余保持自洽）。

    h_i = sha256( h_{i-1} + canonical(record_i) )，h_0 = "0"*64

canonical 规则：按 AUDIT_REQUIRED_FIELDS 顺序取字段，缺失即 "∅"（占位符而非跳过
—— 跳过会让"删字段"与"原值就是空"产生同一哈希，蓄意者可借此抹除字段）。
"""
from __future__ import annotations

import hashlib
import json
from dataclasses import dataclass
from typing import Iterable, List, Mapping, Sequence

from chiyu_kernel.contract import (
    AUDIT_GENESIS_HASH,
    AUDIT_HASH_ALGO,
    AUDIT_REQUIRED_FIELDS,
)

EMPTY_PLACEHOLDER = "∅"


def canonical(record: Mapping, fields: Sequence[str] = AUDIT_REQUIRED_FIELDS) -> str:
    payload = {f: record.get(f, EMPTY_PLACEHOLDER) for f in fields}
    return json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def chain_hash(prev_hash: str, record: Mapping) -> str:
    if AUDIT_HASH_ALGO != "sha256":
        raise ValueError(f"未支持的审计哈希算法：{AUDIT_HASH_ALGO}")
    h = hashlib.sha256()
    h.update(prev_hash.encode("utf-8"))
    h.update(canonical(record).encode("utf-8"))
    return h.hexdigest()


def build_chain(records: Iterable[Mapping]) -> List[str]:
    prev = AUDIT_GENESIS_HASH
    out: List[str] = []
    for r in records:
        prev = chain_hash(prev, r)
        out.append(prev)
    return out


@dataclass(frozen=True)
class VerifyResult:
    ok: bool
    checked: int
    first_bad_index: int = -1
    reason: str = ""


def verify_chain(records: Sequence[Mapping], expected: Sequence[str]) -> VerifyResult:
    """重算并比对。返回首个失配位置（-1 表示全链通过）。"""
    if len(records) != len(expected):
        return VerifyResult(False, min(len(records), len(expected)), -1,
                            f"长度不一致：records={len(records)} expected={len(expected)}")
    prev = AUDIT_GENESIS_HASH
    for i, r in enumerate(records):
        prev = chain_hash(prev, r)
        if prev != expected[i]:
            return VerifyResult(False, i + 1, i, f"第 {i} 条哈希失配")
    return VerifyResult(True, len(records))

"""canary 探针件（N7）—— 决策落库三判据的**统一判定式**（foundation 单一事实源）。

⛔ 为什么要在基座 ship 这个：三个消费方各自自造探针 ⇒ 判据口径漂移
   （"count>0 就算落库"这类恒绿探针曾是本仓真实事故）。本模块把三判据
   的**判定式**收归基座维护，消费方只注入接线（store + 决策回调）。

三判据（缺一不可，任一不满足即探针失败）：
  1. **决策真落库**：event 行存在 **且** particle 行存在（只查 event 不查
     particle 挡不住"半写"—— 两张表须同事务成对出现）；
  2. **记忆网真物化**：`graph_edge` 中该决策的 `edge_source='engine'` 边数
     **恰好等于**注入 references 的非空谓词数（多写=幻影边，少写=漏物化）；
  3. **红线挡落库**：红线决策回调的结果不落库且被审批闸拦下
     （REJECTED 或 PENDING；注入了红线回调才可判定，否则该判据为 False）。

接口说明（与 enterprise 反馈 #N7 原案 `canary_probe(dsn, profile)` 的差异，诚实标注）：
  取 `store`（鸭子类型：`execute` / `execute_fetch` / `schema`）而非 `dsn` ——
  消费方（如 APS 用 psycopg3）已持有自己的连接层，注入 store 免内核绑死驱动
  （与 R9 驱动解耦同一取向）；dsn 场景由消费方先 `PgVectorStore(profile)` 自建。
  `run_decision` / `run_redline_decision` 为消费方接线的决策回调
  （内核不得反向依赖 chiyu_decision，分层铁律）。
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Any, Callable, Mapping, Optional, Sequence

__all__ = ["CanaryReport", "canary_probe"]


@dataclass(frozen=True)
class CanaryReport:
    """探针结论：三判据逐项布尔 + 细节（排障用，不含敏感值）。"""

    decision_persisted: bool
    memory_net_materialized: bool
    redline_blocked: bool
    details: Mapping[str, Any] = field(default_factory=dict)

    @property
    def ok(self) -> bool:
        """三判据全绿才算通过（部分绿 = 未通过，禁"主要判据绿"式放水）。"""
        return self.decision_persisted and self.memory_net_materialized \
            and self.redline_blocked


def _q(schema: str, table: str) -> str:
    return f"{schema}.{table}" if schema else table


def _table_of(sql: str, schema: str, table: str) -> bool:
    """判定 SQL 是否查询 `<schema>.<table>`（供打桩 store 路由读请求）。"""
    pat = re.compile(rf"FROM\s+{re.escape(_q(schema, table))}\b", re.IGNORECASE)
    return bool(pat.search(sql))


def canary_probe(*, store: Any,
                 run_decision: Callable[[], Any],
                 decision_id: str = "canary-dec-1",
                 references: Optional[Mapping[str, Sequence[str]]] = None,
                 run_redline_decision: Optional[Callable[[], Any]] = None) -> CanaryReport:
    """对已接线的决策链路执行三判据。**只读核验**，自身不写库。

    参数：
      store               消费方 store（PgVectorStore 或同鸭子类型；须有
                          `schema` / `execute_fetch`）
      run_decision        消费方决策回调：执行一次**正常**决策
                          （run_spine + persister 已接到同一 store），返回
                          SpineResult 同形对象（须有 `persisted` 属性）
      decision_id         本轮回调使用的 decision_id（须与回调内一致）
      references          记忆网供给（键=谓词、值=端点列表）；None/空 ⇒
                          判据 2 不可判定（False，禁静默放行）
      run_redline_decision 红线决策回调：执行一次**红线命中**的决策；
                          None ⇒ 判据 3 不可判定（False）
    """
    schema = getattr(store, "schema", "")
    details: dict = {}

    # ── 判据 1：决策真落库（event + particle 成对）────────────────────
    result = run_decision()
    details["run_persisted"] = bool(getattr(result, "persisted", False))
    event_row = store.execute_fetch(
        f"SELECT id FROM {_q(schema, 'event')} WHERE id=%s", (decision_id,))
    particle_row = store.execute_fetch(
        f"SELECT particle_id FROM {_q(schema, 'particle')} WHERE particle_id=%s",
        (decision_id,))
    details["event_row"] = event_row is not None
    details["particle_row"] = particle_row is not None
    decision_persisted = (details["run_persisted"] and details["event_row"]
                          and details["particle_row"])

    # ── 判据 2：记忆网真物化（engine 边数 == 非空谓词数，恰好相等）──────
    expected_edges = sum(1 for v in (references or {}).values() if v) \
        if references else 0
    if expected_edges == 0:
        details["memory_net"] = "未提供 references ⇒ 判据 2 不可判定"
        memory_net_materialized = False
    else:
        row = store.execute_fetch(
            f"SELECT count(*) AS n FROM {_q(schema, 'graph_edge')} "
            f"WHERE from_id=%s AND edge_source='engine'", (decision_id,))
        actual = int(row["n"]) if isinstance(row, dict) else int(row[0])
        details["memory_net"] = f"engine 边 {actual} / 期望 {expected_edges}"
        memory_net_materialized = (actual == expected_edges)

    # ── 判据 3：红线挡落库 ────────────────────────────────────────────
    if run_redline_decision is None:
        details["redline"] = "未注入红线决策回调 ⇒ 判据 3 不可判定"
        redline_blocked = False
    else:
        r2 = run_redline_decision()
        rejected = getattr(r2, "approval_rejected", None) is not None
        pending = bool(getattr(r2, "pending", False))
        not_persisted = not getattr(r2, "persisted", False)
        details["redline"] = (f"persisted={getattr(r2, 'persisted', None)}, "
                              f"rejected={rejected}, pending={pending}")
        redline_blocked = not_persisted and (rejected or pending)

    return CanaryReport(
        decision_persisted=decision_persisted,
        memory_net_materialized=memory_net_materialized,
        redline_blocked=redline_blocked,
        details=details,
    )

@dataclass
class SupplyReport:
    required: tuple = ()
    supplied: tuple = ()
    missing: tuple = ()
    degraded_layers: tuple = ()

    @property
    def consistent(self) -> bool:
        return not self.missing

    def as_metrics(self) -> Mapping[str, float]:
        req = len(self.required)
        hit = req - len(self.missing)
        return {
            "context_supply_rate": round(hit / req, 4) if req else 1.0,
            "context_missing": float(len(self.missing)),
            "context_degraded_layers": float(len(self.degraded_layers)),
        }


@dataclass
class AssemblyResult:
    facts: dict = field(default_factory=dict)
    supply_report: SupplyReport = field(default_factory=SupplyReport)


def assemble(needs: Sequence[KnowledgeNeed],
             provided: Mapping[str, Iterable]) -> AssemblyResult:
    """按需装配。

    needs    KnowledgeNeed 序列（需求侧，由调用方按场景声明）
    provided {layer: [{"dim":..., "key":..., "value":...}, ...]}（供给侧）
    返回    facts + supply_report（缺口**结构化**返回，不吞不糊）
    """
    facts: dict = {}
    supplied_keys: list = []
    missing: list = []
    degraded: list = []
    for need in needs:
        rows = list(provided.get(need.layer) or [])
        if not rows:
            degraded.append(need.layer)
        hit = [r for r in rows
               if (not need.dims or _dim_of(r) in need.dims)
               and (not need.key or r.get("key") == need.key)]
        if hit:
            supplied_keys.append((need.layer, need.key))
            for r in hit:
                facts.setdefault(need.layer, []).append(r.get("value"))
        else:
            missing.append((need.layer, need.key))
    return AssemblyResult(
        facts=facts,
        supply_report=SupplyReport(
            required=tuple((n.layer, n.key) for n in needs),
            supplied=tuple(supplied_keys),
            missing=tuple(missing),
            degraded_layers=tuple(degraded),
        ),
    )


def _dim_of(row: Mapping) -> str:
    d = row.get("dim")
    if isinstance(d, dict):
        return d.get("key") or d.get("dim") or ""
    return d or ""

def _clamp_unique(items: Iterable[str], limit: int) -> tuple[str, ...]:
    """去重 + 保序 + 截断（空串丢弃）。需求编码一律走此入口，防「无声膨胀」。"""
    seen: set[str] = set()
    out: list[str] = []
    for raw in items:
        s = str(raw).strip()
        if not s or s in seen:
            continue
        seen.add(s)
        out.append(s)
        if len(out) >= limit:
            break
    return tuple(out)

"""输入层护栏：拦截 NL 中的脚本/注入，从源头杜绝「NL 直出 HTML」。

纵深防御第一层（输入层）。第二层结构层 `validate_page_schema`，第三层渲染层 `render_page`。
"""
from __future__ import annotations

import re
from dataclasses import dataclass

# 注入模式（IGNORECASE）：脚本 / 伪协议 / 事件属性 / 动态求值 / 嵌入对象
_INJECTION_PATTERNS: tuple[re.Pattern[str], ...] = (
    re.compile(r"<\s*script", re.IGNORECASE),
    re.compile(r"javascript\s*:", re.IGNORECASE),
    re.compile(r"<\s*iframe", re.IGNORECASE),
    re.compile(r"<\s*object", re.IGNORECASE),
    re.compile(r"<\s*embed", re.IGNORECASE),
    re.compile(r"on\w+\s*=", re.IGNORECASE),
    re.compile(r"eval\s*\("),
)


@dataclass(frozen=True)
class GuardResult:
    """输入护栏结果。``ok`` 决定可否进入解析层。"""

    ok: bool
    reason: str = ""

    def __bool__(self) -> bool:
        return self.ok


def guard_nl_input(text: str) -> GuardResult:
    """拦截脚本注入。命中任一模式即拒绝，返回字段级原因。"""
    if not isinstance(text, str):
        return GuardResult(False, "输入必须是字符串")
    for pat in _INJECTION_PATTERNS:
        m = pat.search(text)
        if m:
            return GuardResult(False, f"检测到注入模式: {m.group(0)!r}")
    return GuardResult(True)


__all__ = ["GuardResult", "guard_nl_input"]
# ----- FILE: packages/chiyu-portal/src/chiyu_portal/nl_parser.py -----
"""NL 解析层：自然语言 → 受控 Schema。⛔ 绝不产出 HTML（守卫 + 测试双重锁死）。

确定性规则映射（无 LLM 直出）。实体→粒子经注入 ``entity_map``；未识别实体 → 低置信 +
``needsClarification``。三段式中本层只产出 Schema，渲染器的 HTML 是另一层的事。
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Any

from .schema import SCHEMA_VERSION

# 意图词 → 页面类型（缺省 dashboard）。顺序即优先级。
_TYPE_RULES: tuple[tuple[tuple[str, ...], str], ...] = (
    (("工作台", "协作", "workspace"), "workspace"),
    (("审批", "审核", "approval"), "approval"),
    (("表单", "录入", "编辑", "form"), "form"),
    (("详情", "明细", "detail"), "detail"),
    (("表", "列表", "清单", "table"), "table"),
    (("看板", "dashboard", "指标", "总览", "概览"), "dashboard"),
)


@dataclass
class ParseResult:
    """解析结果：结构化 Schema（绝不 HTML）+ 置信度 + 是否需澄清。"""

    schema: dict[str, Any]
    confidence: float
    needs_clarification: bool
    notes: list[str] = field(default_factory=list)

    def to_dict(self) -> dict[str, Any]:
        return {
            "schema": self.schema,
            "confidence": self.confidence,
            "needsClarification": self.needs_clarification,
            "notes": self.notes,
        }


def parse_nl_to_schema(
    text: str,
    *,
    entity_map: dict[str, str] | None = None,
    title: str | None = None,
) -> ParseResult:
    """将自然语言描述解析为受控 Schema（绝不直出 HTML）。

    ``entity_map``：NL 实体词 → 粒子类型（消费方注入）。未命中 → 降置信 + needsClarification。
    """
    entity_map = entity_map or {}
    notes: list[str] = []

    schema_type = "dashboard"
    for keywords, t in _TYPE_RULES:
        if any(k in text for k in keywords):
            schema_type = t
            break

    particle_type: str | None = None
    for token, pt in entity_map.items():
        if token and token in text:
            particle_type = pt
            break

    confidence = 0.9
    needs = False
    if particle_type is None:
        confidence = 0.3
        needs = True
        notes.append("未识别实体，无法确定粒子类型")

    highlight = _detect_threshold(text)

    components: list[dict[str, Any]] = []
    if particle_type is not None:
        metric: dict[str, Any] = {
            "field": "value", "agg": "count", "label": "计数",
        }
        comp: dict[str, Any] = {
            "kind": "metric-card",
            "title": title or f"{schema_type} 视图",
            "dataBinding": {
                "source": "particle",
                "particleType": particle_type,
                "metrics": [metric],
            },
        }
        if highlight is not None:
            comp["style"] = {"highlight": highlight}
        components.append(comp)

    schema: dict[str, Any] = {
        "version": SCHEMA_VERSION,
        "type": schema_type,
        "title": title or f"{schema_type} 视图",
        "layout": {"columns": 2, "theme": "light"},
        "components": components,
    }
    return ParseResult(
        schema=schema, confidence=confidence,
        needs_clarification=needs, notes=notes,
    )


def _detect_threshold(text: str) -> dict[str, Any] | None:
    """阈值高亮：低于|小于|少于 X% → lt；高于|超过|大于|多于 X% → gt（红色）。"""
    m = re.search(r"(低于|小于|少于)\s*(\d+(?:\.\d+)?)\s*%?", text)
    if m:
        return {"field": "value", "op": "lt", "value": float(m.group(2)) / 100.0, "color": "red"}
    m = re.search(r"(高于|超过|大于|多于)\s*(\d+(?:\.\d+)?)\s*%?", text)
    if m:
        return {"field": "value", "op": "gt", "value": float(m.group(2)) / 100.0, "color": "red"}
    return None

"""页面生命周期：护栏 → 解析 → 校验 → 落库 draft → 发布。

无 PG 可全绿（内存存储，便于测试；消费方可替换为真实存储）。
三段式全链路在此编排：``guard_nl_input`` → ``parse_nl_to_schema`` → ``validate_page_schema``。
"""
from __future__ import annotations

import uuid
from dataclasses import dataclass, field
from typing import Any

from .guardrails import GuardResult, guard_nl_input
from .nl_parser import ParseResult, parse_nl_to_schema
from .renderer import RenderResult, render_page
from .schema import ValidationError, validate_page_schema


@dataclass
class PageDraft:
    """页面草稿（未发布）。"""

    draft_id: str
    schema: dict[str, Any]
    confidence: float
    needs_clarification: bool
    notes: list[str] = field(default_factory=list)
    published: bool = False


@dataclass
class CreateResult:
    """创建结果。``ok=False`` 时携带护栏原因或字段级校验错误。"""

    ok: bool
    draft: PageDraft | None = None
    errors: tuple[ValidationError, ...] = ()
    guard_reason: str = ""

    def __bool__(self) -> bool:
        return self.ok


class PageStore:
    """内存页面存储（无 PG 依赖）。消费方应替换为真实持久层。"""

    def __init__(self) -> None:
        self._drafts: dict[str, PageDraft] = {}

    def create_page_from_nl(
        self,
        text: str,
        *,
        particle_types: Any,
        entity_map: dict[str, str] | None = None,
        allowed_actions=None,
        status_fields: Any = (),
        snapshot_fields: Any = (),
    ) -> CreateResult:
        guard: GuardResult = guard_nl_input(text)
        if not guard:
            return CreateResult(False, guard_reason=guard.reason)

        parsed: ParseResult = parse_nl_to_schema(text, entity_map=entity_map)
        # 实体未识别（低置信 / needsClarification）→ 无粒子绑定，不应落草稿
        if parsed.needs_clarification:
            return CreateResult(
                False, errors=(ValidationError("nl", "实体未识别，需澄清后再生成页面"),),
            )
        errors = validate_page_schema(
            parsed.schema, particle_types=particle_types,
            allowed_actions=allowed_actions, status_fields=status_fields,
            snapshot_fields=snapshot_fields,
        )
        if errors:
            return CreateResult(False, errors=errors)

        draft = PageDraft(
            draft_id=uuid.uuid4().hex,
            schema=parsed.schema,
            confidence=parsed.confidence,
            needs_clarification=parsed.needs_clarification,
            notes=parsed.notes,
        )
        self._drafts[draft.draft_id] = draft
        return CreateResult(True, draft=draft)

    def publish_page(self, draft_id: str) -> bool:
        draft = self._drafts.get(draft_id)
        if draft is None:
            return False
        draft.published = True
        return True

    def render(self, draft_id: str, **kwargs: Any) -> RenderResult:
        draft = self._drafts.get(draft_id)
        if draft is None:
            return RenderResult(False, errors=(ValidationError("draft", "草稿不存在"),))
        return render_page(draft.schema, **kwargs)

# ----- FILE: packages/chiyu-portal/src/chiyu_portal/renderer.py -----
"""渲染层：受控 Schema → HTML。⛔ 唯一渲染出口；渲染前再校验；动态值全转义；
交互仅 data-action 声明式；输出强制无 <script> / on*=。纵深防御第三层。

⛔ 判别力：非法 Schema 直接拒绝（返回字段级错误），绝不尽力渲染。
"""
from __future__ import annotations

import html
import re
from dataclasses import dataclass
from typing import Any, Iterable

from .schema import SCHEMA_VERSION, ValidationError, validate_page_schema

_SCRIPT_RE = re.compile(r"<\s*script", re.IGNORECASE)
_ONATTR_RE = re.compile(r"\son\w+\s*=", re.IGNORECASE)


@dataclass
class RenderResult:
    """渲染结果。``ok=False`` 时携带字段级错误（非法 Schema 被拒）。"""

    ok: bool
    html: str = ""
    errors: tuple[ValidationError, ...] = ()

    def __bool__(self) -> bool:
        return self.ok


def render_page(
    schema: dict[str, Any],
    *,
    particle_types: Any,
    allowed_actions: Iterable[str] | None = None,
    status_fields: Iterable[str] = (),
    snapshot_fields: Iterable[str] = (),
) -> RenderResult:
    """渲染受控 Schema 为 HTML。非法 Schema 拒绝（字段级错误），绝不尽力渲染。"""
    errors = validate_page_schema(
        schema, particle_types=particle_types,
        allowed_actions=allowed_actions, status_fields=status_fields,
        snapshot_fields=snapshot_fields,
    )
    if errors:
        return RenderResult(False, errors=errors)

    out: list[str] = []
    out.append("<!DOCTYPE html>")
    out.append('<html lang="zh"><head><meta charset="utf-8">')
    out.append("<title>" + html.escape(str(schema.get("title", ""))) + "</title>")
    out.append('<link rel="stylesheet" href="/portal/page.css">')
    out.append("</head><body>")
    out.append('<main class="portal-page">')
    out.append('<h1 class="page-title">' + html.escape(str(schema.get("title", ""))) + "</h1>")
    for comp in schema.get("components", []) or []:
        out.append(_render_component(comp))
    out.append("</main></body></html>")

    rendered = "".join(out)
    # 渲染层硬闸门：不得出现 script / 内联事件属性
    if _SCRIPT_RE.search(rendered) or _ONATTR_RE.search(rendered):
        return RenderResult(False, errors=(ValidationError("render", "渲染产物含禁用标记"),))
    return RenderResult(True, html=rendered)


def _render_component(comp: dict[str, Any]) -> str:
    kind = html.escape(str(comp.get("kind", "")))
    title = html.escape(str(comp.get("title", "")))
    parts: list[str] = [f'<section class="component component-{kind}">']
    parts.append(f'<h2 class="component-title">{title}</h2>')
    binding = comp.get("dataBinding")
    if isinstance(binding, dict):
        pt = html.escape(str(binding.get("particleType", "")))
        parts.append(f'<div class="binding" data-particle-type="{pt}">')
        for m in binding.get("metrics", []) or []:
            if not isinstance(m, dict):
                continue
            label = html.escape(str(m.get("label", "")))
            field = html.escape(str(m.get("field", "")))
            parts.append(f'<span class="metric" data-field="{field}">{label}</span>')
        parts.append("</div>")
    for act in comp.get("actions", []) or []:
        if not isinstance(act, dict):
            continue
        label = html.escape(str(act.get("label", "")))
        action = html.escape(str(act.get("action", "")))
        # 交互仅声明式 data-action，无 onclick / 内联 JS
        parts.append(f'<button class="action-btn" data-action="{action}">{label}</button>')
    parts.append("</section>")
    return "".join(parts)


def render_conversation(
    *,
    events: Any = None,
    composer_action: str = "send_message",
    goal: str = "",
) -> "RenderResult":
    """渲染 dsh 式对话页：session 日志 + 目标栏 + composer。

    声明式 ``data-action``，无内联 JS；复用渲染层硬闸门（无 <script> / on*=）。
    ``events`` 形如 ``[{"role": "user"|"assistant", "content": ...}]`` 或
    ``[{"kind": "user/message"|"assistant/message", "payload": {"content": ...}}]``
    （兼容 SessionEvent 形态，零领域字面量）。
    """
    out: list[str] = []
    out.append("<!DOCTYPE html>")
    out.append('<html lang="zh"><head><meta charset="utf-8">'
               '<title>对话</title><link rel="stylesheet" href="/portal/page.css"></head><body>')
    out.append('<main class="portal-page conversation-page">')
    if goal:
        out.append(f'<div class="goal-bar">{html.escape(goal)}</div>')
    out.append('<section class="component component-chat-log">'
               '<h2 class="component-title">对话</h2><div class="chat-log">')
    for ev in events or []:
        role = html.escape(str(ev.get("role", ev.get("kind", ""))))
        payload = ev.get("payload") or {}
        content = html.escape(str(ev.get("content", payload.get("content", ""))))
        out.append(f'<div class="msg msg-{role}">'
                   f'<span class="role">{role}</span>'
                   f'<span class="text">{content}</span></div>')
    out.append('</div></section>')
    out.append(f'<form class="composer" data-action="{html.escape(composer_action)}">'
               '<input name="message" class="composer-input" placeholder="输入消息…" '
               'autocomplete="off">'
               '<button type="submit" class="action-btn">发送</button></form>')
    out.append('</main></body></html>')

    rendered = "".join(out)
    if _SCRIPT_RE.search(rendered) or _ONATTR_RE.search(rendered):
        return RenderResult(False, errors=(ValidationError("render", "渲染产物含禁用标记"),))
    return RenderResult(True, html=rendered)

"""chiyu-portal 受控页面 Schema 协议（零领域字面量）。

三段式：NL → 受控 Schema → 运行时渲染器。本模块是 **Schema 层**（结构化中间表示），
定义协议常量与校验器。渲染器（`renderer.py`）是唯一渲染出口，且渲染前会再次调用
`validate_page_schema` —— 非法 Schema 必须被拒（不是「尽力渲染」）。

⛔ 粒子类型 / 动作经**注入**，框架层零硬编码业务字面量。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Iterable

from chiyu_kernel.contract.particle import ParticleTypeSet

# ── 协议常量（形态，无业务取值）────────────────────────────────────────
PAGE_TYPES: tuple[str, ...] = (
    "dashboard", "table", "form", "detail", "workspace", "approval",
    "conversation",
)
COMPONENT_KINDS: tuple[str, ...] = (
    "metric-card", "table", "form", "detail",
    "reasoning-trace", "approval-queue", "conflict-marker",
    "goal-form", "interaction-panel", "acceptance-checklist",
    "chat-log", "composer", "goal-bar",
)
THEMES: tuple[str, ...] = ("light", "dark")
FILTER_OPS: tuple[str, ...] = ("eq", "ne", "lt", "gt", "lte", "gte", "in")
AGG_FUNCS: tuple[str, ...] = ("count", "sum", "avg", "min", "max", "latest")

# 框架级默认动作白名单（消费方注入其完整 ActionRegistry 覆盖）
DEFAULT_ALLOWED_ACTIONS: tuple[str, ...] = (
    "data-particle-read", "data-particle-write", "page-reload", "aiattr-generate",
)

SCHEMA_VERSION = "0.2"


@dataclass(frozen=True)
class ValidationError:
    """字段级校验错误（路径定位，便于前端逐字段回显）。"""

    path: str
    message: str

    def __str__(self) -> str:
        return f"{self.path}: {self.message}"


def validate_page_schema(
    schema: dict[str, Any],
    *,
    particle_types: ParticleTypeSet,
    allowed_actions: Iterable[str] | None = None,
    status_fields: Iterable[str] = (),
    snapshot_fields: Iterable[str] = (),
) -> tuple[ValidationError, ...]:
    """校验受控页面 Schema。返回字段级错误元组（空 = 通过）。

    ⛔ 判别力自证要点（均须被拒，非静默映射）：
      - 未知 particleType / 页面类型 / 组件类型
      - 非白名单 action
      - 非法过滤算子 / 聚合函数
      - 状态字段非 eq 或聚合
      - 快照字段非 latest
    """
    if not isinstance(schema, dict):
        return (ValidationError("schema", "Schema 必须是对象"),)

    errors: list[ValidationError] = []

    version = schema.get("version")
    if version is not None and version != SCHEMA_VERSION:
        errors.append(ValidationError("version", f"不支持的 version: {version!r}"))

    ptype = schema.get("type")
    if ptype is not None and ptype not in PAGE_TYPES:
        errors.append(ValidationError("type", f"未知页面类型: {ptype!r}"))

    layout = schema.get("layout")
    if isinstance(layout, dict):
        theme = layout.get("theme")
        if theme is not None and theme not in THEMES:
            errors.append(ValidationError("layout.theme", f"未知主题: {theme!r}"))

    actions = set(allowed_actions) if allowed_actions is not None else set(DEFAULT_ALLOWED_ACTIONS)
    status_set = set(status_fields)
    snapshot_set = set(snapshot_fields)

    for i, comp in enumerate(_iter_components(schema)):
        cpath = f"components[{i}]"
        kind = comp.get("kind")
        if kind is not None and kind not in COMPONENT_KINDS:
            errors.append(ValidationError(f"{cpath}.kind", f"未知组件类型: {kind!r}"))

        binding = comp.get("dataBinding")
        if isinstance(binding, dict):
            _validate_binding(
                binding, f"{cpath}.dataBinding", errors,
                particle_types, status_set, snapshot_set,
            )

        for j, act in enumerate(comp.get("actions", []) or []):
            apath = f"{cpath}.actions[{j}]"
            aname = act.get("action") if isinstance(act, dict) else None
            if aname is not None and aname not in actions:
                errors.append(ValidationError(f"{apath}.action", f"非白名单动作: {aname!r}"))

    return tuple(errors)


def _iter_components(schema: dict[str, Any]) -> list[dict]:
    comps = schema.get("components")
    return comps if isinstance(comps, list) else []


def _validate_binding(
    binding: dict[str, Any], path: str, errors: list[ValidationError],
    particle_types: ParticleTypeSet, status_set: set[str], snapshot_set: set[str],
) -> None:
    src = binding.get("source")
    if src is not None and src != "particle":
        errors.append(ValidationError(f"{path}.source", f"仅支持 particle 数据源: {src!r}"))

    pt = binding.get("particleType")
    if pt is not None and pt not in particle_types.names:
        errors.append(ValidationError(f"{path}.particleType", f"未知粒子类型: {pt!r}（受控枚举）"))

    for fi, f in enumerate(binding.get("filters", []) or []):
        if not isinstance(f, dict):
            continue
        fpath = f"{path}.filters[{fi}]"
        op = f.get("op")
        field_name = f.get("field")
        if op is not None and op not in FILTER_OPS:
            errors.append(ValidationError(f"{fpath}.op", f"非法过滤算子: {op!r}"))
        if field_name in status_set and op is not None and op != "eq":
            errors.append(ValidationError(f"{fpath}.op", f"状态字段仅支持 eq: {field_name!r}"))

    for mi, m in enumerate(binding.get("metrics", []) or []):
        if not isinstance(m, dict):
            continue
        mpath = f"{path}.metrics[{mi}]"
        agg = m.get("agg")
        field_name = m.get("field")
        if agg is not None and agg not in AGG_FUNCS:
            errors.append(ValidationError(f"{mpath}.agg", f"非法聚合函数: {agg!r}"))
        if field_name in status_set and agg is not None:
            errors.append(ValidationError(f"{mpath}.agg", f"状态字段不可聚合: {field_name!r}"))
        if field_name in snapshot_set and agg is not None and agg != "latest":
            errors.append(ValidationError(f"{mpath}.agg", f"快照字段仅支持 latest: {field_name!r}"))

"""chiyu decide —— 跑决策脊柱（确定性分支，零 IO、不接 LLM）。

⛔ 诚实原则：八要素未提供的项一律留空，由脊柱如实报 missing；
   绝不伪造齐备以换取「看起来跑通」。
"""
from __future__ import annotations

import argparse
import json
from typing import Any, Dict

from chiyu_decision.api import run_spine
from chiyu_kernel.contract.decision import RUBRIC_KEYS
from ._plugins import resolve

DEFAULT_SCORE = 3          # RUBRIC_SCALE_MAX=4；全 3 ⇒ ratio=27/36=0.75 < 0.79 阈值
DEFAULT_THRESHOLD = 0.79


def _parse_scores(raw: str | None) -> Dict[str, float]:
    """解析 `--scores clarity=3,logic=4`；未提及的键取 DEFAULT_SCORE。"""
    scores: Dict[str, float] = {k: float(DEFAULT_SCORE) for k in RUBRIC_KEYS}
    if not raw:
        return scores
    for item in raw.split(","):
        item = item.strip()
        if not item:
            continue
        if "=" not in item:
            raise ValueError(f"评分项须形如 key=value：{item!r}")
        key, _, value = item.partition("=")
        key = key.strip()
        if key not in RUBRIC_KEYS:
            raise ValueError(f"未知尺子键 {key!r}；合法键：{', '.join(RUBRIC_KEYS)}")
        scores[key] = float(value.strip())
    return scores


def _build_elements(topic: str) -> Dict[str, object]:
    """八要素：只填真实提供的项（intent = topic）。

    ⛔ 不得用空串/占位补齐其余键 —— `check_elements` 只判**键是否存在**，
       补空键会让「要素缺失」恒为「无」，属典型假齐备。CLI 只拿到了议题，
       故如实只给 intent，其余由脊柱报 missing。
    """
    return {"intent": topic}


def handler(args: argparse.Namespace) -> int:
    scores = _parse_scores(args.scores)
    elements = _build_elements(args.topic)
    mode, enabled, disabled, mwarn = resolve(args.mode)
    result = run_spine(
        context={
            "decision_id": args.decision_id or f"cli-{args.topic[:24]}",
            "topic": args.topic,
            "work_mode": mode.id,        # 观测面：本次走的工作模式
        },
        profile=None,                  # ⛔ 无默认值，预览模式显式传 None
        rubric_scores=scores,
        elements=elements,
        persister=None,                # 不落库
        threshold=args.threshold,
    )
    payload: Dict[str, Any] = {
        "decision_id": result.decision_id,
        "stage_reached": result.stage_reached,
        "confidence": round(result.confidence, 4),
        "autonomous": result.autonomous,
        "redline_hit": result.redline_hit,
        "redline_reasons": list(result.redline_reasons),
        "rubric_missing": list(result.rubric_missing),
        "elements_missing": list(result.elements_missing),
        "persisted": result.persisted,
        "work_mode": mode.id,
        "work_mode_plugins": list(enabled),
    }
    if args.json:
        print(json.dumps(payload, ensure_ascii=False, indent=2))
        return 0
    print(f"decision_id   : {payload['decision_id']}")
    print(f"stage_reached : {payload['stage_reached']}")
    print(f"confidence    : {payload['confidence']}  (阈值 {args.threshold})")
    print(f"autonomous    : {payload['autonomous']}")
    print(f"redline_hit   : {payload['redline_hit']} {payload['redline_reasons'] or ''}")
    print(f"尺子缺失      : {', '.join(payload['rubric_missing']) or '无'}")
    print(f"要素缺失      : {', '.join(payload['elements_missing']) or '无'}")
    print(f"工作模式      : {payload['work_mode']}（启用插件：{', '.join(payload['work_mode_plugins']) or '无'}）")
    return 0


def register(sub: Any) -> None:
    p = sub.add_parser("decide", help="跑决策脊柱（确定性分支）")
    p.add_argument("--topic", required=True, help="决策议题描述")

    p.add_argument("--threshold", type=float, default=DEFAULT_THRESHOLD, help="自主放行阈值")
    p.add_argument("--decision-id", default=None, help="指定决策 ID")
    p.add_argument("--mode", default=None,
                   help="工作模式：standard/ptc/minimal/creative（缺省=standard）")
    p.add_argument("--json", action="store_true", help="机器可读输出")
    p.set_defaults(handler=handler)

def _namespace() -> Dict[str, Any]:
    import chiyu_kernel.api as K
    import chiyu_kernel.contract as C
    import chiyu_decision.api as D
    import chiyu_portal.api as P
    from chiyu_kernel.contract.particle import CORE_ROLES, ParticleTypeSet

    def render(nl: str, out: str | None = None) -> str:
        guard = P.guard_nl_input(nl)
        if not guard.ok:
            return f"护栏拒绝：{guard.reason}"
        parsed = P.parse_nl_to_schema(nl)
        res = P.render_page(parsed.schema, particle_types=ParticleTypeSet(names=CORE_ROLES))
        if not res.ok:
            return "渲染拒绝：" + "; ".join(str(e) for e in res.errors)
        if out:
            Path(out).write_text(res.html, encoding="utf-8")
            return f"已写出 {out}"
        return res.html

    def readiness() -> str:
        return K.readiness_report().summary

    def decide(topic: str) -> Any:
        return D.run_spine(
            context={"topic": topic}, profile=None,
            rubric_scores={}, elements={"intent": topic}, persister=None,
        )

# ----- FILE: packages/chiyu-cli/src/chiyu_cli/commands/portal.py -----
"""chiyu portal render —— NL → 受控 Schema → HTML（零 Web 服务，落盘即可看）。

链路：guard_nl_input（注入护栏）→ parse_nl_to_schema（绝不直出 HTML）
     → render_page（唯一渲染出口，非法 Schema 字段级拒绝）。
"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path
from typing import Any, Dict, List

from chiyu_kernel.contract.particle import CORE_ROLES, ParticleTypeSet
from chiyu_portal.api import guard_nl_input, parse_nl_to_schema, render_page, render_onboarding
from ._plugins import resolve


def _parse_entity_map(raw: str | None) -> Dict[str, str]:
    """解析 `--entity-map 客户=entity,订单=entity`（NL 实体词 → 粒子类型）。

    ⛔ 映射值由**调用方注入** —— 内核零领域字面量，CLI 不得内置任何业务实体词。
    """
    mapping: Dict[str, str] = {}
    if not raw:
        return mapping
    for item in raw.split(","):
        item = item.strip()
        if not item:
            continue
        if "=" not in item:
            raise ValueError(f"实体映射项须形如 实体词=粒子类型：{item!r}")
        token, _, ptype = item.partition("=")
        mapping[token.strip()] = ptype.strip()
    return mapping


def _particle_set(raw: str | None, entity_map: Dict[str, str]) -> ParticleTypeSet:
    """默认核心角色 + `--particle-types` + 实体映射值（后者否则 Schema 校验会拒）。"""
    extra: List[str] = []
    if raw:
        extra += [x.strip() for x in raw.split(",") if x.strip()]
    extra += [p for p in entity_map.values() if p]
    # ⚠ 必须对 CORE_ROLES + extra 整体去重：映射值可能与核心角色重名（如 entity）
    names: tuple = tuple(dict.fromkeys((*CORE_ROLES, *extra)))
    return ParticleTypeSet(names=names)


def handler(args: argparse.Namespace) -> int:
    guard = guard_nl_input(args.nl)
    if not guard.ok:
        print(f"[护栏拒绝] {guard.reason}", file=sys.stderr)   # ⚠ 字段名是 reason（单数）
        return 2

    mode, enabled, disabled, mwarn = resolve(args.mode)
    # 极简模式隐藏门户导航：强制 portal=False，仅产出裸页面
    portal_on = bool(mode.orchestration.get("portal", True))
    if mwarn:
        for w in mwarn:
            print(f"[模式警告] {w}", file=sys.stderr)

    try:
        entity_map = _parse_entity_map(args.entity_map)
        particle_types = _particle_set(args.particle_types, entity_map)
    except ValueError as exc:
        print(f"[参数非法] {exc}", file=sys.stderr)
        return 2

    parsed = parse_nl_to_schema(args.nl, entity_map=entity_map, title=args.title)
    print(f"schema_type   : {parsed.schema.get('type')}")
    print(f"confidence    : {parsed.confidence}")
    print(f"需澄清        : {parsed.needs_clarification}")
    if parsed.notes:
        print(f"notes         : {'; '.join(parsed.notes)}")
    print(f"工作模式      : {mode.id}（portal导航={'开' if portal_on else '关'}；"
          f"启用插件：{', '.join(enabled) or '无'}）")

    result = render_page(parsed.schema, particle_types=particle_types)
    if not result.ok:
        print("[渲染拒绝] 非法 Schema，字段级错误：", file=sys.stderr)
        for err in result.errors:
            print(f"  - {err}", file=sys.stderr)
        return 2

    if args.out:
        path = Path(args.out)
        if path.parent != Path(""):
            path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(result.html, encoding="utf-8")
        print(f"已写出        : {path.resolve()}  ({len(result.html)} 字节)")
    else:
        print(result.html)
    return 0


def _onboarding_handler(args: argparse.Namespace) -> int:

    else:
        print(result.html)

"""chiyu selftest —— 调 pytest 跑指定包测试（退出码透传）。"""
from __future__ import annotations

import argparse
import subprocess
import sys
from pathlib import Path
from typing import Any, List

# 本文件位于 <仓库根>/packages/chiyu-cli/src/chiyu_cli/commands/selftest.py
# parents: [0]=commands [1]=chiyu_cli [2]=src [3]=chiyu-cli [4]=packages [5]=仓库根
_ROOT = Path(__file__).resolve().parents[5]

_PACKAGES: dict = {
    "kernel": ("packages/chiyu-kernel/tests",),
    "decision": ("packages/chiyu-decision",),
    "portal": ("packages/chiyu-portal",),
    "cli": ("packages/chiyu-cli",),
}


def _targets(name: str) -> List[str]:
    if name == "all":
        return ["packages"]
    return [str(_ROOT / p) for p in _PACKAGES[name]]


def handler(args: argparse.Namespace) -> int:
    cmd = [
        sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider",
        *_targets(args.package),
    ]
    if args.keyword:
        cmd += ["-k", args.keyword]
    print("$ " + " ".join(cmd))
    completed = subprocess.run(cmd, cwd=str(_ROOT))
    return int(completed.returncode)


def register(sub: Any) -> None:
    p = sub.add_parser("selftest", help="跑包测试")
    p.add_argument("--package", default="all",
                   choices=("all", "kernel", "decision", "portal", "cli"), help="目标包")
    p.add_argument("-k", dest="keyword", default=None, help="pytest -k 过滤")
    p.set_defaults(handler=handler)

"""chiyu 命令派发入口。

⛔ 本模块只做派发与参数装配；业务一律下沉到 `commands/*`。
"""
from __future__ import annotations

import argparse
import importlib.util
import sys
from typing import Any, List, Optional

from . import __version__, shell
from .commands import decide, evolution, portal, selftest, modes


def build_parser() -> argparse.ArgumentParser:
    """构造顶层解析器并注册全部子命令。"""
    parser = argparse.ArgumentParser(
        prog="chiyu",
        description="ChiYu AI 原生决策平台 —— 终端运行入口",
    )
    parser.add_argument("--version", action="version", version=f"chiyu {__version__}")
    sub = parser.add_subparsers(dest="command", required=True)

    decide.register(sub)
    portal.register(sub)
    evolution.register(sub)
    selftest.register(sub)
    modes.register(sub)
    shell.register(sub)

    # `dev`（完整 Python REPL）仅 full 版存在；external 构建不打包该模块。
    # ⛔ 这是构建形态探测（external 版无 dev.pyc），不是运行期动态派发。
    if importlib.util.find_spec("chiyu_cli.commands.dev") is not None:
        from .commands import dev

        dev.register(sub)

    return parser


def main(argv: Optional[List[str]] = None) -> int:
    """CLI 主入口。返回进程退出码。"""
    parser = build_parser()
    args: Any = parser.parse_args(argv)   # required=True ⇒ 必有 handler
    return int(args.handler(args) or 0)


if __name__ == "__main__":  # pragma: no cover
    sys.exit(main())
# ----- FILE: packages/chiyu-cli/src/chiyu_cli/shell.py -----
"""受限命令壳 —— ⛔ 白名单外一律拒绝（技术上封死 raw import 逃逸）。

external（外部顾问版）只提供本壳，不提供完整 Python REPL；
因此外部顾问**无法**借 `chiyu shell` 执行 `import chiyu_kernel` 读取源码。
"""
from __future__ import annotations

import argparse
import shlex
import sys
from typing import Any, List

from .commands.decide import handler as _decide
from .commands.evolution import handler as _readiness
from .commands.portal import handler as _render
from .commands.selftest import handler as _selftest
from .commands._plugins import resolve

HELP = """可用命令：
  decide "议题"                       跑决策脊柱
  render "页面描述" [--out p.html]     NL → HTML
  readiness [--assert]                开闸前置认证
  selftest [kernel|decision|portal|cli|all]
  help                                显示本帮助
  exit                                退出
⛔ 仅以上命令可用；不支持任意 Python 语句与 import。
"""

_EXIT = -1          # 哨兵：退出循环


def _run(argv: List[str]) -> int:
    """把壳内一行解析为受控调用。未知命令 ⇒ 拒绝并退 2。"""
    if not argv:
        return 0
    head, rest = argv[0], argv[1:]

    if head == "decide":
        topic = rest[0] if rest else ""
        if not topic:
            print('用法：decide "议题"', file=sys.stderr)
            return 2
        return int(_decide(argparse.Namespace(
            topic=topic, scores=None, threshold=0.79, decision_id=None, json=False)))

    if head == "render":
        nl = rest[0] if rest else ""
        if not nl:
            print('用法：render "页面描述" [--out p.html]', file=sys.stderr)
            return 2
        out = None
        if "--out" in rest:
            idx = rest.index("--out")
            out = rest[idx + 1] if idx + 1 < len(rest) else None
        return int(_render(argparse.Namespace(
            nl=nl, out=out, title=None, particle_types=None)))

    if head == "readiness":
        return int(_readiness(argparse.Namespace(
            assert_="--assert" in rest, json=False)))

    if head == "selftest":
        pkg = rest[0] if rest else "all"
        return int(_selftest(argparse.Namespace(package=pkg, keyword=None)))

    if head == "help":
        print(HELP)
        return 0

    if head == "exit":
        return _EXIT

    print(f"[拒绝] 不支持的命令 {head!r}；输入 help 查看可用命令。", file=sys.stderr)
    return 2


def run_shell(mode_id: str | None = None) -> int:
    """启动受限命令壳。返回 0（正常退出）。"""
    mode, enabled, disabled, mwarn = resolve(mode_id)
    if mwarn:
        for w in mwarn:
            print(f"[模式警告] {w}", file=sys.stderr)
    banner = (f"chiyu shell（受限命令壳）— 工作模式：{mode.id}（{mode.label}）；"
              f"启用插件：{', '.join(enabled) or '无'}。输入 help 查看命令，exit 退出。")
    if mode.orchestration.get("portal_hidden"):
        banner += "  极简模式：门户导航已隐藏。"
    print(banner)
    while True:
        try:
            line = input("chiyu> ")
        except (EOFError, KeyboardInterrupt):
            print()
            return 0
        try:
            argv = shlex.split(line)
        except ValueError as exc:
            print(f"[拒绝] 无法解析：{exc}", file=sys.stderr)
            continue
        rc = _run(argv)
        if rc == _EXIT:
            return 0
    return 0


def handler(args: Any) -> int:
    return run_shell(getattr(args, "mode", None))


def register(sub: Any) -> None:
    p = sub.add_parser("shell", help="受限命令壳（两版共有）")
    p.add_argument("--mode", default=None,
                   help="工作模式：standard/ptc/minimal/creative（缺省=standard）")
    p.set_defaults(handler=handler)

## §1 分层与命名（一个决定，影响后面全部）

新增内核子包 **`chiyu_kernel.evolution`**（自进化层）。

**为什么不塞进 `orchestration`**：`orchestration` 回答「任务怎么跑」，
`evolution` 回答「**平台改自己**」。二者失败语义相反——任务失败是业务损失，
自变失败是**平台把标尺改错**。混在一个包里，G3（只读护栏）的边界就划不清。

```
chiyu_kernel/
├─ contract/evolution.py      # T1  L0 契约（零依赖、零领域值）
├─ evolution/
│   ├─ knob.py                # T2  参数白名单 + 风险分档 + G1
│   ├─ replay.py              # T3  影子重放
│   ├─ prescription.py        # T4  处方状态机（事务原子 + 挂起 + 回滚）
│   ├─ skill_draft.py         # T6  SKILL 草稿闸 + 命名空间隔离
│   └─ guard.py               # T7  G2/G3 护栏（红线挂起 + 只读路径）
└─ orchestration/
    ├─ agent_registry.py      # T5  Agent 注册期拒绝
    └─ idem.py                # T8  幂等键（补内核缺口）
```

⛔ **零领域字面量铁律（本层最易犯）**：`evolution` 层**不得**出现具体参数取值、
具体红线条目、具体 SKILL 名。内核只提供**判据形态**，全部取值由消费方注入：

| 内容 | 注入通道 |
|---|---|
| 参数白名单与取值区间 | `KnobTable`（消费方构造后传入） |
| 红线目录 | `TenantProfile.redline`（已有 `RedlineSpec`） |
| SKILL 允许命名空间 | `profile.extras["skill_namespaces"]` |
| 唯一例外：`ai-` 基线前缀 | 写在 `contract/evolution.py` —— 它是**护栏判据**，不是领域值（理由见 T6） |

### T2 · `evolution/knob.py` —— 参数白名单 + 风险分档 + G1

**目标**：解决「哪些参数能被自动改」与「放宽永不自动」。

```python
"""运行参数表（白名单 + 风险分档）—— 照抄 dispatch_cfg 的三条硬化。

⛔ 为什么白名单外键一律忽略
   配置面 ≠ 隐式接口面。若允许"配置里写了就认"，夜批就能靠新增键给自己造权。

⛔ 为什么方向必须声明而不能内核推断
   "阈值变小"在 autonomous_threshold 上是放宽，在 max_inflight 上也是放宽，
   但在 cooldown_days 上是**收紧**。推断会错，错了就是 R1 自我扩权。
   故由 KnobSpec.widen_direction 显式声明：值朝哪边走算放宽。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Mapping, Optional, Tuple

from chiyu_kernel.contract import (
    DIR_FLAT, DIR_TIGHTEN, DIR_WIDEN, TIER_AUTO, TIER_FORBIDDEN, TIER_GUARDED,
)


class KnobError(ValueError):
    """参数表契约违规（非法键 / 越界 / 未声明方向）⇒ 抛，不静默。"""


@dataclass(frozen=True)
class KnobSpec:
    name: str
    tier: str = TIER_AUTO
    min_value: Optional[float] = None
    max_value: Optional[float] = None
    # 值朝哪个方向变化算"放宽自主权"；None = 与自主权无关（永不被判 widen）
    widen_direction: Optional[str] = None   # "down" | "up" | None
    # 阈值类参数：命中 G1 ⇒ 放宽永不自动
    autonomy_related: bool = False

    def __post_init__(self) -> None:
        if self.tier not in (TIER_AUTO, TIER_GUARDED, TIER_FORBIDDEN):
            raise KnobError(f"未知风险档：{self.tier!r}")
        if self.widen_direction not in (None, "down", "up"):
            raise KnobError(f"widen_direction 须为 down/up/None：{self.widen_direction!r}")
        if self.autonomy_related and self.widen_direction is None:
            raise KnobError(
                f"autonomy_related 的参数必须声明 widen_direction：{self.name!r}"
                "（否则 G1 无法判定放宽方向 ⇒ 自我扩权不可检出）")

    def clamp_check(self, value: Any) -> Tuple[bool, str]:
        try:
            v = float(value)
        except (TypeError, ValueError):
            return False, f"非数值：{value!r}"
        if self.min_value is not None and v < self.min_value:
            return False, f"低于下限 {self.min_value}：{v}"
        if self.max_value is not None and v > self.max_value:
            return False, f"高于上限 {self.max_value}：{v}"
        return True, ""


class KnobTable:
    """参数白名单（消费方构造）。外键一律忽略并留痕 `rejected`。"""

    def __init__(self, specs) -> None:
        self._by_name: Dict[str, KnobSpec] = {}
        for s in specs:
            if s.name in self._by_name:
                raise KnobError(f"重复注册参数：{s.name!r}")
            self._by_name[s.name] = s

    def read(self, cfg: Mapping[str, Any]) -> Tuple[Dict[str, Any], List[str]]:
        """(生效值, rejected)。非法即回落 spec 缺省（不抛错、不就地裁剪）。"""
        out: Dict[str, Any] = {}
        rejected: List[str] = []
        for name, spec in self._by_name.items():
            if name in cfg:
                ok, why = spec.clamp_check(cfg[name])
                if ok:
                    out[name] = cfg[name]
                    continue
                rejected.append(f"{name}: {why}")
            # 未配或非法 ⇒ 不加默认（消费方未声明缺省则留空，由调用方兜底）
        unknown = [k for k in cfg if k not in self._by_name]
        for k in sorted(unknown):
            rejected.append(f"{k}: 非白名单键，已忽略")
        return out, rejected

    def has(self, name: str) -> bool:
        return name in self._by_name

    def spec(self, name: str) -> KnobSpec:
        if name not in self._by_name:
            raise KnobError(f"非白名单参数：{name!r}（配置面 ≠ 接口面）")
        return self._by_name[name]


def classify_change(spec: KnobSpec, old: Any, new: Any) -> str:
    """判定变更方向：widen / tighten / flat（G1 的判据核心）。"""
    try:
        o, n = float(old), float(new)
    except (TypeError, ValueError):
        return DIR_FLAT
    if o == n or spec.widen_direction is None:
        return DIR_FLAT
    if spec.widen_direction == "down":
        return DIR_WIDEN if n < o else DIR_TIGHTEN
    return DIR_WIDEN if n > o else DIR_TIGHTEN


def is_forbidden_change(spec: KnobSpec, direction: str) -> bool:
    """G1：自主权相关参数的**放宽**永不自动。"""
    return bool(spec.autonomy_related and direction == DIR_WIDEN)


__all__ = ["KnobError", "KnobSpec", "KnobTable", "classify_change", "is_forbidden_change"]
```

**判据**：
- 外键被忽略且进 `rejected`；非法值回落且不抛错；
- `autonomy_related=True` 却未声明 `widen_direction` ⇒ **构造期抛错**（把 R1 拦在构造期，而非运行期）；
- `classify_change` 对 `widen_direction="down"` 的参数：变小 ⇒ `widen` ⇒ `is_forbidden_change` 为真。

### T3 · `evolution/replay.py` —— 影子重放

**目标**：消灭 blind patch（§12.3 第 1 条）。

```python
"""影子重放 —— 应用前用历史样本 dry-run。

⛔ 为什么"无样本"必须判 reject 而不是 pass
   没有重放样本的变更 = blind patch：线上劣化只能靠事后发现。若空缺时返回
   "通过"，守卫就成了摆设——夜批在没有样本的那些天恰好会全量盲改。

⛔ 为什么 formula 必须是**同一个函数对象**
   "语义相近的重写版本"在边界上必然分叉（阈值取等、空值处理），而分叉只会在
   线上样本上暴露。故重放不接受字符串公式，只接受与运行引擎同源的可调用对象。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Callable, Iterable, List, Mapping, Sequence

from chiyu_kernel.contract import FAIL_SAFE_VERDICT, VERDICT_PASS, VERDICT_REJECT


@dataclass(frozen=True)
class ReplayVerdict:
    verdict: str
    checked: int = 0
    mismatches: List[str] = field(default_factory=list)
    reason: str = ""

    @property
    def ok(self) -> bool:
        return self.verdict == VERDICT_PASS


def shadow_replay(
    samples: Sequence[Mapping[str, Any]],
    formula: Callable[[Mapping[str, Any]], Any],
    new_params: Mapping[str, Any],
    *,
    min_samples: int = 1,
) -> ReplayVerdict:
    """用 `new_params` 代入 `formula` 重算历史样本，与样本自带期望值比对。

    samples 每条须含 `__expected`（该样本当时的运行结论）。缺 `__expected`
    的样本**计入 mismatch**（不是跳过——跳过会让"无基线"伪装成"全通过"）。
    """
    if not callable(formula):
        return ReplayVerdict(VERDICT_REJECT, reason="formula 不可调用 ⇒ 拒绝（禁字符串公式）")
    n = len(samples)
    if n < min_samples:
        return ReplayVerdict(
            VERDICT_REJECT, checked=n,
            reason=f"样本量不足（{n} < {min_samples}）⇒ 无重放即 blind patch，拒绝出方")

    bad: List[str] = []
    for i, s in enumerate(samples):
        if "__expected" not in s:
            bad.append(f"#{i}: 缺 __expected 基线（不跳过）")
            continue
        try:
            got = formula({k: v for k, v in s.items() if k != "__expected"} | dict(new_params))
        except Exception as exc:                      # 异常 ⇒ 保守拒绝（fail-safe）
            bad.append(f"#{i}: 重算异常 {type(exc).__name__}: {exc}")
            continue
        if got != s["__expected"] and not _close(got, s["__expected"]):
            bad.append(f"#{i}: 期望 {s['__expected']!r} 实得 {got!r}")

    if bad:
        return ReplayVerdict(VERDICT_REJECT, checked=n, mismatches=bad,
                             reason=f"{len(bad)}/{n} 条重放失配")
    return ReplayVerdict(VERDICT_PASS, checked=n)


def _close(a: Any, b: Any, *, tol: float = 1e-9) -> bool:
    try:
        return abs(float(a) - float(b)) <= tol
    except (TypeError, ValueError):
        return False


__all__ = ["ReplayVerdict", "shadow_replay"]
```

**判据**：`samples=[]` ⇒ `verdict=reject`（**这条最关键**，空跑通过是假绿）；
`formula` 非可调用 ⇒ reject；缺 `__expected` ⇒ 计入失配而非跳过。

### T4 · `evolution/prescription.py` —— 处方状态机（事务原子 + 挂起 + 回滚）

**目标**：消灭「配置已改、处方仍待批」半应用态（§12.3 第 3 条）。

```python
"""处方状态机 —— 自变动作的**唯一**落盘路径。

⛔ 为什么应用必须事务原子
   配置写入与处方状态迁移若分两步，任一中间失败都留下"配置已改、处方仍待批"
   的半应用态：运行态被污染，且**无法自动收敛**（状态机不知道该往前还是往后）。

⛔ 为什么要有 verdict 字段
   纯方法论执行默认不抛异常，用"没抛错"判成功是假阳性。故落盘必须带
   机器可读 verdict，空值/异常一律按 reject 挂起（fail-safe）。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Callable, List, Mapping, Optional, Tuple

from chiyu_kernel.contract import (
    DIR_WIDEN, FAIL_SAFE_VERDICT, PRESCRIPTION_STATES, PRESCRIPTION_TRANSITIONS,
    VERDICT_PASS, VERDICT_REJECT,
)


class PrescriptionError(ValueError):
    """处方契约违规（非法转换 / 事务半应用 / 缺 verdict）⇒ 抛。"""


@dataclass(frozen=True)
class Prescription:
    prescription_id: str
    knob: str
    old_value: Any
    new_value: Any
    direction: str
    tier: str
    state: str = "pending"
    verdict: str = FAIL_SAFE_VERDICT      # 默认保守：没写结论 = 拒绝
    replay_ref: Optional[str] = None
    audit_ref: Optional[str] = None
    reason: str = ""

    def __post_init__(self) -> None:
        if self.state not in PRESCRIPTION_STATES:
            raise PrescriptionError(f"未知处方状态：{self.state!r}")
        if not self.prescription_id:
            raise PrescriptionError("prescription_id 必填（回滚要靠它定位）")


def assert_prescription_transition(frm: str, to: str) -> None:
    if frm not in PRESCRIPTION_STATES:
        raise PrescriptionError(f"未知源状态：{frm!r}")
    if to not in PRESCRIPTION_STATES:
        raise PrescriptionError(f"未知目标状态：{to!r}")
    if to not in PRESCRIPTION_TRANSITIONS[frm]:
        raise PrescriptionError(
            f"非法处方转换：{frm} → {to}（允许 {sorted(PRESCRIPTION_TRANSITIONS[frm])}）")


def apply_atomic(
    rx: Prescription,
    *,
    writer: Callable[[str, Any], None],
    on_state: Callable[[str, str], None],
    rollback_writer: Optional[Callable[[str, Any], None]] = None,
) -> Tuple[Prescription, str]:
    """事务原子应用：先写配置，再迁状态；任一步抛错 ⇒ 逆写回滚并置 rolled_back。

    ⛔ writer / on_state 必须都能抛：吞掉异常的 writer 会让本函数"看上去成功"。
       故约定：writer 内部**不得**捕获异常后静默返回。
    """
    if rx.verdict != VERDICT_PASS:
        return _suspend(rx, on_state, reason=f"verdict={rx.verdict!r} ⇒ 挂起（fail-safe）"), "suspended"
    if rx.direction == DIR_WIDEN:
        return _suspend(rx, on_state, reason="G1：放宽自主权永不自动 ⇒ 挂起"), "suspended"

    assert_prescription_transition(rx.state, "auto_applied")
    try:
        writer(rx.knob, rx.new_value)                 # ① 写配置
        on_state(rx.prescription_id, "auto_applied")  # ② 迁状态（同一事务）
    except Exception as exc:
        # 半应用态的唯一补救：立刻逆写，不留"改了一半"
        if rollback_writer is not None:
            try:
                rollback_writer(rx.knob, rx.old_value)
            except Exception as rb:
                raise PrescriptionError(
                    f"应用失败且回滚失败（需人工介入）：{exc} / {rb}") from rb
        raise PrescriptionError(f"应用失败已回滚：{exc}") from exc
    return Prescription(**{**rx.__dict__, "state": "auto_applied"}), "auto_applied"


def roll_back(
    rx: Prescription,
    *,
    writer: Callable[[str, Any], None],
    on_state: Callable[[str, str], None],
) -> Prescription:
    """一键回滚 = 逆动作，**同样留痕**（回滚本身也是一次自变）。"""
    assert_prescription_transition(rx.state, "rolled_back")
    writer(rx.knob, rx.old_value)
    on_state(rx.prescription_id, "rolled_back")
    return Prescription(**{**rx.__dict__, "state": "rolled_back"})


def _suspend(rx: Prescription, on_state, *, reason: str) -> Prescription:
    assert_prescription_transition(rx.state, "suspended")
    on_state(rx.prescription_id, "suspended")
    return Prescription(**{**rx.__dict__, "state": "suspended", "reason": reason})


__all__ = ["PrescriptionError", "Prescription", "assert_prescription_transition",
           "apply_atomic", "roll_back"]
```

**判据**：`verdict != pass` 或 `direction == widen` ⇒ 挂起（**不写配置**）；
`writer` 成功而 `on_state` 抛错 ⇒ 逆写回滚且抛 `PrescriptionError`；
`roll_back` 从 `auto_applied` 出发（非法转换抛错）。

### T5 · `orchestration/agent_registry.py` —— Agent 注册期拒绝

**目标**：新 agent 必携契约才能注册（§12.2 对象①）。

```python
"""Agent 注册表（自变对象①）—— 注册期拒绝，范式照抄 `actions/registry.py`。

⛔ 为什么必须是注册期而不是事后审计
   demo `agent/registry.py` 是静态 dict、无闸 ⇒ "加一个 agent"不报错，
   事后审计才发现它没契约/没成功标准。事后发现意味着它**已经跑过了**。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, List, Mapping, Optional, Tuple


class AgentRegistryError(ValueError):
    """Agent 注册契约违规 ⇒ 注册期抛错（非事后审计）。"""


@dataclass(frozen=True)
class AgentSpec:
    key: str
    contract_task_id: str
    skills: Tuple[str, ...] = ()
    memory: Tuple[str, ...] = ()
    knowledge_scope: Mapping = field(default_factory=dict)
    success: str = ""

    def missing(self) -> List[str]:
        out = []
        if not self.key:
            out.append("key")
        if not self.contract_task_id:
            out.append("contract_task_id")
        if not self.skills:
            out.append("skills")
        if not self.memory:
            out.append("memory")
        if not self.success.strip():
            out.append("success")
        return out


class AgentRegistry:
    def __init__(self) -> None:
        self._by_key: Dict[str, AgentSpec] = {}
        self._by_task: Dict[str, str] = {}

    def register(self, spec: AgentSpec) -> None:
        # ① 必携完整契约（缺一即拒）
        miss = spec.missing()
        if miss:
            raise AgentRegistryError(
                f"agent {spec.key!r} 契约不完整（缺 {miss}）⇒ 拒绝注册"
                "（无成功标准的 agent 跑完了也无法判定成败）")
        # ② 重复 key
        if spec.key in self._by_key:
            raise AgentRegistryError(f"重复注册 agent：{spec.key!r}")
        # ③ 契约键冲突（一个契约键只能被一个 agent 持有）
        owner = self._by_task.get(spec.contract_task_id)
        if owner is not None:
            raise AgentRegistryError(
                f"契约键 {spec.contract_task_id!r} 已由 {owner!r} 持有，"
                f"新增 {spec.key!r} ⇒ 应并入既有 agent，不得另起")
        self._by_key[spec.key] = spec
        self._by_task[spec.contract_task_id] = spec.key

    def get(self, key: str) -> Optional[AgentSpec]:
        return self._by_key.get(key)

    def keys(self) -> Tuple[str, ...]:
        return tuple(sorted(self._by_key))

    def audit(self) -> List[str]:
        """返回 errors（非空 ⇒ 注册表不合规）。"""
        return [f"agent {k} success 非判定式（无可执行结论）"
                for k, s in self._by_key.items() if len(s.success.strip()) < 8]


__all__ = ["AgentRegistryError", "AgentSpec", "AgentRegistry"]
```

**判据**：缺 `success` 的 agent **注册期抛错**；重复 key / 契约键冲突抛错（`pytest.raises` 实测）。

### T6 · `evolution/skill_draft.py` —— SKILL 草稿闸 + 命名空间隔离

**目标**：自生成 SKILL 不得自动生效、不得写入 `ai-*` 基线（10 大能力不可动）。

```python
"""SKILL 草稿闸（自变对象②）—— 命名空间隔离 + 所调 Action 必须存在。

⛔ 为什么 `ai-` 前缀写死在内核而不是可配
   10 大能力是审计可复现的基线。若消费方可覆盖该前缀，护栏 G3 即被"配置"绕过——
   这种绕过每一步看起来都合法（改的只是配置项）。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Iterable, List, Mapping, Tuple

from chiyu_kernel.contract import BASELINE_SKILL_PREFIX, VERDICT_PASS, VERDICT_REJECT


class SkillDraftError(ValueError):
    """草稿违规 ⇒ 抛（草稿绝不静默生效）。"""


@dataclass(frozen=True)
class SkillDraft:
    name: str
    calls_actions: Tuple[str, ...] = ()
    status: str = "draft"          # draft → validated → 待交付
    namespace: str = "method"      # 默认新命名空间，非 ai-*

    def fq_name(self) -> str:
        return f"{self.namespace}-{self.name}"


def validate_draft(draft: SkillDraft, *, action_names: Iterable[str],
                   allowed_namespaces: Iterable[str] = ("method", "domain")) -> Tuple[str, List[str]]:
    """返回 (verdict, errors)。errors 为空且 verdict=pass 才允许继续。"""
    errs: List[str] = []
    if draft.namespace == BASELINE_SKILL_PREFIX.rstrip("-") or \
       draft.fq_name().startswith(BASELINE_SKILL_PREFIX):
        errs.append(
            f"⛔ 自生成 SKILL 不得写入基线命名空间 {BASELINE_SKILL_PREFIX}*"
            f"（10 大能力不得新增/删除/重编号）")
    if allowed_namespaces and draft.namespace not in tuple(allowed_namespaces):
        errs.append(f"命名空间 {draft.namespace!r} 不在允许集 {tuple(allowed_namespaces)} 内")
    known = set(action_names)
    for a in draft.calls_actions:
        if a not in known:
            errs.append(f"所调 Action 未注册：{a!r}（草稿不得引用不存在的 Action）")
    if not draft.calls_actions:
        errs.append("草稿未声明所调 Action ⇒ 无法校验（空草稿按拒绝）")
    return (VERDICT_REJECT if errs else VERDICT_PASS), errs


def promote(draft: SkillDraft, *, verdict: str) -> SkillDraft:
    """草稿晋升。**verdict != pass ⇒ 抛**（草稿绝不自动生效的机器形式）。"""
    if verdict != VERDICT_PASS:
        raise SkillDraftError(f"草稿 {draft.fq_name()!r} 未通过校验 ⇒ 禁止晋升")
    if draft.status != "draft":
        raise SkillDraftError(f"仅 draft 可晋升（当前 {draft.status!r}）")
    return SkillDraft(**{**draft.__dict__, "status": "validated"})


__all__ = ["SkillDraftError", "SkillDraft", "validate_draft", "promote"]
```

**判据**：`namespace="ai"` 且 `name="xxx"` ⇒ `fq_name()="ai-xxx"` ⇒ 命中基线前缀 ⇒ 拒绝；
`calls_actions` 含未注册 Action ⇒ 拒绝；`promote(verdict=reject)` ⇒ 抛。

# G3：自主写入通道对这些前缀**物理只读**（不可配置、不可豁免）

### T8 · `orchestration/idem.py` —— 幂等键（**补内核缺口**）

```python
"""幂等键 —— 夜批防重跑（内核此前**无此原语**，§12.3 的"沿用"实为未验证假设）。

⛔ 为什么必须有
   B 档夜间无人值守：重跑一次夜批 ⇒ 重复出方 + 重复应用。配置被改第二遍时
   old_value 已被覆盖 ⇒ 回滚链断裂。这不是"多跑一次"，是**无声的配置漂移**。
"""
from __future__ import annotations

import hashlib
import json
from typing import Mapping, Sequence


def derive_idem_key(scope: str, parts: Sequence[Mapping] | Sequence[str]) -> str:
    """由 scope + 内容派生稳定键。内容相同 ⇒ 键相同（跨进程、跨天稳定）。"""
    s = (scope or "").strip()
    if not s:
        raise ValueError("idem scope 必填（空 scope 会让不同批次撞同一个键）")
    payload = json.dumps(
        [p if isinstance(p, str) else {k: p[k] for k in sorted(p)} for p in parts],
        ensure_ascii=False, sort_keys=True, separators=(",", ":"), default=str,
    )
    h = hashlib.sha256()
    h.update(s.encode("utf-8"))
    h.update(b"|")
    h.update(payload.encode("utf-8"))
    return f"idem:{s}:{h.hexdigest()[:16]}"


def nightly_idem_key(reason: str, *, window: str, parts: Sequence[Mapping] | Sequence[str]) -> str:
    """夜批专用：`nightly:<reason>@<window>`。window 进键 ⇒ 换天自然换键。"""
    if not (reason or "").strip():
        raise ValueError("nightly reason 必填（否则不同原因的夜批互相覆盖）")
    return derive_idem_key(f"nightly:{reason}@{window}", list(parts))


__all__ = ["derive_idem_key", "nightly_idem_key"]
```

**判据**：同内容两次调用键相同；`scope=""` 抛；换 `window` ⇒ 键不同。

## §0 目标与非目标

**目标**：新增第四包 `packages/chiyu-cli`，提供 `chiyu` 终端命令，两种交付形态：
- **full（全代码版）**：`chiyu shell` 受限命令壳 + `chiyu dev` 完整 Python REPL；三兄弟包以 `.py` 源码分发。
- **external（外部顾问版）**：仅受限命令壳，**无 `dev` 命令**；四包以 **pyc-only 轮子**分发（磁盘上零 `.py`）。

**非目标**（守边界）：
- ❌ 不接真实 LLM（决策走确定性分支）
- ❌ 不新增粒子类型 / 不改业务域模型（守 §10 硬约束）
- ❌ 不加 Web 服务 / 不补 `page.css`
- ❌ 不污染内核：CLI 只 import `chiyu_kernel.api` + `chiyu_kernel.contract.*`（与兄弟包同款，符合分层铁律）

| 入口 | 精确签名 |

| 决策 | `run_spine(*, context: Mapping, profile, rubric_scores: Mapping[str,float], elements: Mapping[str,object], persister=None, threshold=0.79, redline_items=()) -> SpineResult`<br>⚠ `profile` **无默认值**，预览模式须显式传 `profile=None` |

（兄弟包 `chiyu-portal/schema.py:14` 与 `chiyu-decision/scoring.py:12` 就是这么取的，符合「消费方只许 import contract 与 api」。）

```python
"""chiyu 命令派发入口。"""
from __future__ import annotations

import argparse
import importlib.util
import sys

from .commands import decide, evolution, portal, selftest, shell


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="chiyu",
        description="ChiYu AI 原生决策平台 —— 终端运行入口",
    )
    parser.add_argument("--version", action="version", version=f"chiyu {__version__}")
    sub = parser.add_subparsers(dest="command", required=True)

    decide.register(sub)
    portal.register(sub)
    evolution.register(sub)
    selftest.register(sub)
    shell.register(sub)

    # `dev`（完整 Python REPL）仅 full 版存在；external 构建不打包该模块。
    if importlib.util.find_spec("chiyu_cli.commands.dev") is not None:
        from .commands import dev
        dev.register(sub)

    return parser


def main(argv: list[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    return int(args.handler(args) or 0)


if __name__ == "__main__":  # pragma: no cover
    sys.exit(main())
```

## T2 · `chiyu decide`

**新建 `src/chiyu_cli/commands/decide.py`**

```python
"""chiyu decide —— 跑决策脊柱（确定性分支，零 IO、不接 LLM）。"""
from __future__ import annotations

import argparse
import json
from typing import Any

from chiyu_decision.api import run_spine
from chiyu_kernel.contract.decision import EIGHT_ELEMENTS, RUBRIC_KEYS

DEFAULT_SCORE = 3        # RUBRIC_SCALE_MAX=4；全 3 ⇒ ratio=27/36=0.75 < 0.79 阈值
DEFAULT_THRESHOLD = 0.79


def _parse_scores(raw: str | None) -> dict[str, float]:
    """解析 `--scores clarity=3,logic=4`；未提及的键取 DEFAULT_SCORE。"""
    scores: dict[str, float] = {k: float(DEFAULT_SCORE) for k in RUBRIC_KEYS}
    if not raw:
        return scores
    for item in raw.split(","):
        item = item.strip()
        if not item:
            continue
        if "=" not in item:
            raise ValueError(f"评分项须形如 key=value：{item!r}")
        key, _, value = item.partition("=")
        key = key.strip()
        if key not in RUBRIC_KEYS:
            raise ValueError(f"未知尺子键 {key!r}；合法键：{', '.join(RUBRIC_KEYS)}")
        scores[key] = float(value.strip())
    return scores


def _build_elements(topic: str) -> dict[str, object]:
    """八要素：intent 取 topic，其余留占位 ⇒ 缺失由脊柱如实报出（不伪造齐备）。"""
    elements: dict[str, object] = {"intent": topic}
    for key in EIGHT_ELEMENTS:
        elements.setdefault(key, "")
    return elements


def handler(args: argparse.Namespace) -> int:
    scores = _parse_scores(args.scores)
    elements = _build_elements(args.topic)
    result = run_spine(
        context={"decision_id": args.decision_id or f"cli-{args.topic[:24]}", "topic": args.topic},
        profile=None,                      # ⛔ 无默认值，预览模式显式传 None
        rubric_scores=scores,
        elements=elements,
        persister=None,                    # 不落库
        threshold=args.threshold,
    )
    payload: dict[str, Any] = {
        "decision_id": result.decision_id,
        "stage_reached": result.stage_reached,
        "confidence": round(result.confidence, 4),
        "autonomous": result.autonomous,
        "redline_hit": result.redline_hit,
        "redline_reasons": list(result.redline_reasons),
        "rubric_missing": list(result.rubric_missing),
        "elements_missing": list(result.elements_missing),
        "persisted": result.persisted,
    }
    if args.json:
        print(json.dumps(payload, ensure_ascii=False, indent=2))
    else:
        print(f"decision_id   : {payload['decision_id']}")
        print(f"stage_reached : {payload['stage_reached']}")
        print(f"confidence    : {payload['confidence']}  (阈值 {args.threshold})")
        print(f"autonomous    : {payload['autonomous']}")
        print(f"redline_hit   : {payload['redline_hit']} {payload['redline_reasons'] or ''}")
        print(f"尺子缺失      : {', '.join(payload['rubric_missing']) or '无'}")
        print(f"要素缺失      : {', '.join(payload['elements_missing']) or '无'}")
    return 0


def register(sub: Any) -> None:
    p = sub.add_parser("decide", help="跑决策脊柱（确定性分支）")
    p.add_argument("--topic", required=True, help="决策议题描述")
    p.add_argument("--scores", default=None, help="九尺子评分，如 clarity=4,logic=3")
    p.add_argument("--threshold", type=float, default=DEFAULT_THRESHOLD, help="自主放行阈值")
    p.add_argument("--decision-id", default=None, help="指定决策 ID")
    p.add_argument("--json", action="store_true", help="机器可读输出")
    p.set_defaults(handler=handler)
```

```python
"""chiyu portal render —— NL → 受控 Schema → HTML（零 Web 服务，落盘即可看）。"""
from __future__ import annotations

import argparse
from pathlib import Path
from typing import Any

from chiyu_kernel.contract.particle import CORE_ROLES, ParticleTypeSet
from chiyu_portal.api import guard_nl_input, parse_nl_to_schema, render_page


def _particle_set(raw: str | None) -> ParticleTypeSet:
    extra: tuple[str, ...] = ()
    if raw:
        extra = tuple(x.strip() for x in raw.split(",") if x.strip())
    return ParticleTypeSet(names=(*CORE_ROLES, *extra))


def handler(args: argparse.Namespace) -> int:
    guard = guard_nl_input(args.nl)
    if not guard.ok:
        print(f"[护栏拒绝] {guard.reason}", file=sys.stderr)   # ⚠ 字段名 reason
        return 2

    parsed = parse_nl_to_schema(args.nl, title=args.title)
    print(f"schema_type   : {parsed.schema.get('type')}")
    print(f"confidence    : {parsed.confidence}")
    print(f"需澄清        : {parsed.needs_clarification}")
    if parsed.notes:
        print(f"notes         : {'; '.join(parsed.notes)}")

    result = render_page(parsed.schema, particle_types=_particle_set(args.particle_types))
    if not result.ok:
        print("[渲染拒绝] 非法 Schema，字段级错误：", file=sys.stderr)
        for err in result.errors:
            print(f"  - {err}", file=sys.stderr)
        return 2

    if args.out:
        path = Path(args.out)
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(result.html, encoding="utf-8")
        print(f"已写出        : {path.resolve()}  ({len(result.html)} 字节)")
    else:
        print(result.html)
    return 0


def register(sub: Any) -> None:
    p = sub.add_parser("portal", help="免代码门户")
    sub2 = p.add_subparsers(dest="portal_cmd", required=True)
    r = sub2.add_parser("render", help="NL → HTML")
    r.add_argument("--nl", required=True, help="页面自然语言描述")
    r.add_argument("--out", default=None, help="输出 HTML 路径；缺省打印到 stdout")
    r.add_argument("--title", default=None, help="页面标题")
    r.add_argument("--particle-types", default=None,
                   help=f"追加粒子类型（逗号分隔）；默认仅核心角色 {','.join(CORE_ROLES)}")
    r.set_defaults(handler=handler)
```

```python
"""chiyu evolution readiness —— 自主通道开闸前置认证（§12.3 七闸 + G1/G2/G3）。"""
from __future__ import annotations

import argparse
import json
import sys
from typing import Any

from chiyu_kernel.api import (
    ChannelNotReadyError,
    assert_channel_may_open,
    readiness_report,
)


def handler(args: argparse.Namespace) -> int:
    if args.assert_:
        try:
            report = assert_channel_may_open()
        except ChannelNotReadyError as exc:
            rep = exc.args[0]
            print("[拒绝开闸] 存在阻断闸 RED：", file=sys.stderr)
            for c in rep.blocking_red:
                print(f"  - {c.check_id} {c.title}: {c.detail}", file=sys.stderr)
            return 1
    else:
        report = readiness_report()

    if args.json:
        print(json.dumps(
            [{"check_id": c.check_id, "title": c.title, "status": c.status,
              "detail": c.detail, "discriminating": c.discriminating} for c in report.checks],
            ensure_ascii=False, indent=2))
        return 0

    print(f"{'ID':<14}{'状态':<8}{'有鉴别力':<10}标题")
    print("-" * 72)
    for c in report.checks:
        flag = "是" if c.discriminating else "否"
        print(f"{c.check_id:<14}{c.status:<8}{flag:<10}{c.title}")
        if c.detail:
            print(f"{'':<14}{c.detail}")
    print("-" * 72)
    print(report.summary)
    return 0


def register(sub: Any) -> None:
    p = sub.add_parser("evolution", help="平台自进化")
    sub2 = p.add_subparsers(dest="evolution_cmd", required=True)
    r = sub2.add_parser("readiness", help="开闸前置认证")
    r.add_argument("--assert", dest="assert_", action="store_true",
                   help="未全绿即抛 ChannelNotReadyError 并退 1")
    r.add_argument("--json", action="store_true", help="机器可读输出")
    r.set_defaults(handler=handler)
```

```python
"""chiyu selftest —— 调 pytest 跑指定包测试。"""
from __future__ import annotations

import argparse
import subprocess
import sys
from pathlib import Path
from typing import Any

# 包目录 → 仓库根相对路径（基于本文件位置反推，与 cwd 无关）
_ROOT = Path(__file__).resolve().parents[4]      # .../packages/chiyu-cli/src/chiyu_cli/commands → 仓库根
_PACKAGES: dict[str, tuple[str, ...]] = {
    "kernel": ("packages/chiyu-kernel/tests",),
    "decision": ("packages/chiyu-decision",),
    "portal": ("packages/chiyu-portal",),
    "cli": ("packages/chiyu-cli",),
}


def _targets(name: str) -> list[str]:
    if name == "all":
        return ["packages"]
    return [str(_ROOT / p) for p in _PACKAGES[name]]


def handler(args: argparse.Namespace) -> int:
    cmd = [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", *_targets(args.package)]
    if args.keyword:
        cmd += ["-k", args.keyword]
    print("$ " + " ".join(cmd))
    completed = subprocess.run(cmd, cwd=str(_ROOT))
    return int(completed.returncode)


def register(sub: Any) -> None:
    p = sub.add_parser("selftest", help="跑包测试")
    p.add_argument("--package", default="all",
                   choices=("all", "kernel", "decision", "portal", "cli"), help="目标包")
    p.add_argument("-k", dest="keyword", default=None, help="pytest -k 过滤")
    p.set_defaults(handler=handler)
```

## T6 · 受限命令壳 `shell`（两版共有）

**新建 `src/chiyu_cli/shell.py`**

```python
"""受限命令壳 —— ⛔ 白名单外一律拒绝（技术上封死 raw import 逃逸）。

external 版只提供本壳，不提供完整 Python REPL；因此外部顾问无法借此读源码。
"""
from __future__ import annotations

import shlex

from .commands.decide import handler as _decide
from .commands.evolution import handler as _readiness
from .commands.portal import handler as _render
from .commands.selftest import handler as _selftest

HELP = """可用命令：
  decide "议题" [--scores k=v,...]   跑决策脊柱
  render "页面描述" [--out p.html]    NL → HTML
  readiness [--assert]               开闸前置认证
  selftest [kernel|decision|portal|cli|all]
  help                               显示本帮助
  exit                               退出
⛔ 仅以上命令可用；不支持任意 Python / import。
"""


class _Args:
    """极简 Namespace 替身（避免为壳层重复造 argparse）。"""

    def __init__(self, **kw: object) -> None:
        for k, v in kw.items():
            setattr(self, k, v)


def _run(argv: list[str]) -> int:
    """把壳内一行解析为受控调用。未知命令 ⇒ 拒绝并退 2。"""
    if not argv:
        return 0
    head, rest = argv[0], argv[1:]

    if head == "decide":
        topic = rest[0] if rest else ""
        if not topic:
            print("用法：decide \"议题\"", file=sys.stderr)
            return 2
        return int(_decide(_Args(topic=topic, scores=None, threshold=0.79,
                                 decision_id=None, json=False)))
    if head == "render":
        nl = rest[0] if rest else ""
        if not nl:
            print("用法：render \"页面描述\" [--out p.html]", file=sys.stderr)
            return 2
        out = None
        if "--out" in rest:
            out = rest[rest.index("--out") + 1]
        return int(_render(_Args(nl=nl, out=out, title=None, particle_types=None)))
    if head == "readiness":
        return int(_readiness(_Args(assert_="--assert" in rest, json=False)))
    if head == "selftest":
        pkg = rest[0] if rest else "all"
        return int(_selftest(_Args(package=pkg, keyword=None)))
    if head == "help":
        print(HELP)
        return 0
    if head == "exit":
        return -1                      # 哨兵：退出循环
    print(f"[拒绝] 不支持的命令 {head!r}；输入 help 查看可用命令。", file=sys.stderr)
    return 2


def run_shell() -> int:
    print("chiyu shell（受限命令壳）— 输入 help 查看命令，exit 退出。")
    while True:
        try:
            line = input("chiyu> ")
        except (EOFError, KeyboardInterrupt):
            print()
            return 0
        try:
            argv = shlex.split(line)
        except ValueError as exc:
            print(f"[拒绝] 无法解析：{exc}", file=sys.stderr)
            continue
        rc = _run(argv)
        if rc == -1:
            return 0
    return 0
```

> 需 `import sys`。`_Args` 用 `setattr` 属显式赋值（非动态派发）；若 `make check` 判违规，
> 改为显式 dataclass。

```python
"""chiyu dev —— 完整 Python REPL（⚠ 仅 full 版；external 构建不打包本模块）。

预置三门面与常用快捷函数，供内部开发现场试跑。
"""
from __future__ import annotations

import argparse
import code
from typing import Any

BANNER = """chiyu dev —— 完整 Python REPL（仅全代码版）
预置：K = chiyu_kernel.api / D = chiyu_decision.api / P = chiyu_portal.api
      C = chiyu_kernel.contract *
      render(nl, out=None) · readiness() · decide(topic)
Ctrl-D 退出。"""


def _namespace() -> dict[str, Any]:
    import chiyu_kernel.api as K
    import chiyu_kernel.contract as C
    import chiyu_decision.api as D
    import chiyu_portal.api as P
    from chiyu_kernel.contract.particle import CORE_ROLES, ParticleTypeSet

    def render(nl: str, out: str | None = None) -> str:
        guard = P.guard_nl_input(nl)
        if not guard.ok:
            return f"护栏拒绝：{guard.reason}"
        parsed = P.parse_nl_to_schema(nl)
        res = P.render_page(parsed.schema, particle_types=ParticleTypeSet(names=CORE_ROLES))
        if not res.ok:
            return "渲染拒绝：" + "; ".join(str(e) for e in res.errors)
        if out:
            Path(out).write_text(res.html, encoding="utf-8")
            return f"已写出 {out}"
        return res.html

    def readiness() -> str:
        return K.readiness_report().summary

    def decide(topic: str) -> Any:
        return D.run_spine(context={"topic": topic}, profile=None,
                           rubric_scores={}, elements={"intent": topic}, persister=None)

    return {"K": K, "D": D, "P": P, "C": C, "render": render,
            "readiness": readiness, "decide": decide}


def handler(args: argparse.Namespace) -> int:
    code.interact(banner=BANNER, local=_namespace())
    return 0


def register(sub: Any) -> None:
    p = sub.add_parser("dev", help="完整 Python REPL（仅全代码版）")
    p.set_defaults(handler=handler)
```

> 需 `from pathlib import Path`。`rubric_scores={}` 演示「缺键 ⇒ missing 全报」的诚实行为。

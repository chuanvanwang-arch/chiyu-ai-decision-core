### 0.3 ⛔ T1 必须做对的架构决断：内核零领域字面量

这是整个底座**能否被复用**的成败点。取证：demo `db/schema.py:28-29`

```python
PARTICLE_TYPES = ("cluster", "entity", "place", "decision", "memory", "knowledge", "unstructured_asset")
```

**这七个值里混了两种东西**，必须切开：

| 类别 | 取值 | 归属 | 判据（用户 2026-08-24 立的去污染规则） |
|---|---|---|---|
| **核心语义角色** | `unstructured_asset` / `entity` / `decision` / `memory` / `knowledge` | ✅ **内核** `CORE_ROLES` | 该角色在**任意**领域都成立 |
| **领域对象** | `cluster` / `place` | ❌ **外移 profile** | 群诉把 `place` 当一等对象；APS 把 `SLAB` 当一等对象 |

⛔ **反面例（为什么 `place` 也不能进内核）**：「地点」看似通用，但群诉把 `place` 当一等对象、APS 把 `SLAB` 当一等对象。一旦引入 `place`，下一步就是为 `ticket`/`slab` 开绿灯——内核退化成"群诉专用框架"。

**同理必须外移的还有：**

| 源位置 | 内容 | 处置 |
|---|---|---|
| `db/schema.py:43-56` `K_PREDICATES` P1–P12 | `partOf: unstructured_asset→cluster`、`hasPattern: cluster→knowledge`… | 外移 profile |
| `db/schema.py:59-67` `M_PREDICATES` E1–E7 | `DECIDED_ON` / `OVERRIDES`…（`decision`/`memory`/`knowledge` 为 `to_type`） | ✅ **进内核**（记忆上下文图谱通用），但 `to_type` 为领域值的（`ESTABLISHES_FRAME→cluster`、`CAUSED→cluster`）置为 `None` 由 profile 填 |

```python
"""粒子 / 属性 / 谓词契约 —— L0 单一事实源（零领域字面量）。

⛔ 准入闸（本模块的铁律）
────────────────────────
内核**不得**出现任何具体业务枚举值。判据（2026-08-24 立）：凡出现真实业务系统的
对象名 / 阶段名 / 谓词名，即为污染 → 外移 L4 profile。

本模块只声明**形态**，不声明**取值**：

    进内核（形态）                        外移 L4 profile（取值）
    ────────────────────────────         ────────────────────────────────
    19 种属性类型（通用）                  粒子类型的具体名字（cluster / slab / ticket…）
    边分层 K/M · 边来源 6 值 · 基数 4 值    知识谓词的具体名字（hasPattern / partOf…）
    核心语义角色 5 个                      红线事项目录 · 评分配置
    受控谓词的数据结构

CORE_ROLES 的准入标准：该角色在**任意**业务领域都成立。
    ⛔ 反面例：「地点」看似通用，但群诉以 place 为一等对象、APS 以 SLAB 为一等对象
       —— 一旦纳入 place，下一步就是为 ticket / slab 开绿灯，内核退化为单域框架。
被 `tests/test_contract_no_domain_literals.py` 强制（黑名单见 config/domain-literal-denylist.json）。
"""
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Tuple

# ── 属性类型：19 种有穷集（attio 风格，与具体业务无关）──────────────────
ATTR_TYPES: Tuple[str, ...] = (
    "text", "rich_text", "number", "percent", "currency", "rating",
    "select", "multi_select", "status", "boolean", "timestamp", "date",
    "location", "record_reference", "actor_reference", "interaction",
    "url", "email", "phone",
)

# ── 边与属性的元数据枚举（形态）────────────────────────────────────────
EDGE_LAYERS: Tuple[str, ...] = ("K", "M")
EDGE_SOURCES: Tuple[str, ...] = ("human", "rule", "ai", "llm", "engine", "manual")
ATTR_SOURCES: Tuple[str, ...] = ("human", "rule", "ai", "llm", "engine")
CARDINALITIES: Tuple[str, ...] = ("1:1", "1:N", "N:1", "N:M")

# 属性值槽位（列化存储的五个物理列，禁自由 JSON —— 见 store/ddl.py）
VALUE_SLOTS: Tuple[str, ...] = (
    "value_text", "value_num", "value_bool", "value_ts", "value_set",
)

# 知识分层（粒子归属）
LAYERS: Tuple[str, ...] = ("L1", "L2")

# ── 核心语义角色（跨域恒定，不接受扩展）────────────────────────────────
CORE_ROLES: Tuple[str, ...] = (
    "unstructured_asset",   # 任意域都有非结构化来件（文档 / 工单 / 报告）
    "entity",               # 任意域都有参与主体
    "decision",             # 任意域都有决策记录
    "memory",               # 任意域都有记忆
    "knowledge",            # 任意域都有知识条目
)

# 向量维度：384 补零到 1024（余弦等价）。⛔ 补零是**落库方**的职责，
# 原始维度由 vector/index_vector.py 原样返回（见该模块 triple 契约）。
HASH_DIM = 384
VECTOR_DIM = 1024

_IDENT = re.compile(r"^[a-z][a-z0-9_]*$")


@dataclass(frozen=True)
class ParticleTypeSet:
    """消费方声明的粒子类型全集（L4 取值 + L0 形态约束）。

    ⛔ 必须包含全部 CORE_ROLES —— 否则内核机制（记忆七边依赖 decision/memory/
       knowledge）会在运行期缺锚点，而这属于"配了却没生效"族缺陷（须构造期发现）。
    """

    names: Tuple[str, ...]

    def __post_init__(self) -> None:
        if not self.names:
            raise ValueError("ParticleTypeSet 不得为空")
        if len(set(self.names)) != len(self.names):
            dup = sorted({n for n in self.names if self.names.count(n) > 1})
            raise ValueError(f"粒子类型重复：{dup}")
        bad = [n for n in self.names if not _IDENT.match(n)]
        if bad:
            raise ValueError(f"粒子类型须匹配 [a-z][a-z0-9_]*：{bad}")
        missing = [r for r in CORE_ROLES if r not in self.names]
        if missing:
            raise ValueError(f"粒子类型集缺少核心语义角色：{missing}")

    def sql_check(self, dialect: str = "postgres") -> str:
        """生成 DDL 的 IN 列表。⛔ 由本方法生成，禁止在 store 层手写枚举。"""
        if dialect != "postgres":
            raise ValueError(f"暂不支持的方言：{dialect}")
        return ", ".join(f"'{n}'" for n in self.names)

    def covers(self, role: str) -> bool:
        return role in self.names


@dataclass(frozen=True)
class PredicateSpec:
    """受控谓词的一条定义（形态；取值由 profile 提供）。"""

    predicate: str
    from_type: str
    to_type: str | None
    cardinality: str | None = None
    source: str = "rule"
    layer: str = "K"

    def __post_init__(self) -> None:
        if not _IDENT.match(self.predicate) and not self.predicate.isupper():
            raise ValueError(f"谓词名须为 snake_case 或全大写：{self.predicate!r}")
        if self.layer not in EDGE_LAYERS:

        if self.source not in EDGE_SOURCES:
            raise ValueError(f"source 须 ∈ {EDGE_SOURCES}：{self.source!r}")
        if self.cardinality is not None and self.cardinality not in CARDINALITIES:
            raise ValueError(f"cardinality 须 ∈ {CARDINALITIES}：{self.cardinality!r}")
```

```python
"""契约层零领域字面量守卫（T1 准入闸）。

判据：`contract/` 下所有 .py 的**字符串常量**不得命中黑名单。
为什么必须脚本化：文档条款不会在有人写下 `PARTICLE_TYPES = ("cluster",)` 时变红，
这个测试会。若不守，内核装到第二个项目就会带着第一个项目的领域模型。

⛔ 大驼峰整体词匹配（`\b` 边界）+ 忽略注释/docstring：
   docstring 里为了**说明**「为什么不许出现 cluster」而写出该词是必要的，
   故只检查 `ast.Constant` 的字符串值，不检查注释与 docstring。
"""
from __future__ import annotations

import ast
import json
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[3]          # packages/chiyu-kernel/
CONTRACT = ROOT / "src" / "chiyu_kernel" / "contract"
DENYLIST = Path(__file__).resolve().parents[4] / "config" / "domain-literal-denylist.json"


def _deny_terms() -> list[str]:
    data = json.loads(DENYLIST.read_text(encoding="utf-8"))
    terms: list[str] = []
    for key, values in data.items():
        if key.startswith("_"):
            continue
        terms.extend(values)
    return sorted(set(terms))


def _docstring_nodes(tree: ast.AST) -> set[int]:
    """收集全部 docstring 的节点 id（这些字符串豁免）。"""
    out: set[int] = set()
    for node in ast.walk(tree):
        if isinstance(node, (ast.Module, ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            body = getattr(node, "body", [])
            if body and isinstance(body[0], ast.Expr) and isinstance(body[0].value, ast.Constant):
                if isinstance(body[0].value.value, str):
                    out.add(id(body[0].value))
    return out


def _string_constants(path: Path) -> list[tuple[int, str]]:
    tree = ast.parse(path.read_text(encoding="utf-8"))
    exempt = _docstring_nodes(tree)
    found: list[tuple[int, str]] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Constant) and isinstance(node.value, str):
            if id(node) in exempt:
                continue
            found.append((node.lineno, node.value))
    return found


@pytest.mark.parametrize("path", sorted(CONTRACT.glob("*.py")), ids=lambda p: p.name)
def test_contract_has_no_domain_literals(path: Path):
    terms = _deny_terms()
    hits: list[str] = []
    for lineno, value in _string_constants(path):
        low = value.lower()
        for t in terms:
            if t.isascii() and t.lower() in low:
                hits.append(f"{path.name}:{lineno} 命中领域词 {t!r}（{value!r}）")
            elif not t.isascii() and t in value:
                hits.append(f"{path.name}:{lineno} 命中领域词 {t!r}（{value!r}）")
    assert not hits, (
        "契约层出现领域字面量 ⇒ 底座退化为单域框架：\n  " + "\n  ".join(hits)
```

- [ ] **Step 9: 写 `tools/check_cython_friendly.py`（完整代码，设计 §6.1 的可执行形式）**

```python
"""内核「可编译友好」静态检查（设计 §6.1 五条规范的可执行形式）。

为什么是脚本而不是文档条款：
    文档条款不会在有人写下 `type("X", (), {})` 时变红，这个脚本会，
    且它可被突变证伪（tools/tests/test_check_cython_friendly.py）。

检查项：
    C1 禁运行时造类：`type(...)` 三参数调用、`types.new_class`、`__build_class__`
    C2 禁 **kwargs 动态派发：任何函数的形参含 `**kwargs`
       （理由：Cython 无法为动态关键字派发生成静态绑定，编译期直接失败）
    C3 禁反射取用：`importlib.import_module`、`getattr(obj, <变量>)`
       （`getattr(obj, "常量名")` 允许 —— 它是静态可解析的）
    C4 契约静态声明：`contract/` 下每个模块必须 import dataclasses 或 pydantic
    C5 退出码即闸门：0=通过 / 1=有违规 / 2=用法错误

用法：
    python tools/check_cython_friendly.py packages/
    python tools/check_cython_friendly.py --json packages/
"""
from __future__ import annotations

import argparse
import ast
import json
import sys
from pathlib import Path
from typing import Iterable

CODE_C1 = "C1-runtime-class"
CODE_C2 = "C2-kwargs-dispatch"
CODE_C3 = "C3-reflection"
CODE_C4 = "C4-contract-not-static"

# C3 豁免：这两个模块的**被加载方**需要可独立执行，加载动作发生在 tools/ 内
_REFLECTION_ALLOWLIST: tuple[str, ...] = ("tools/",)


class Finding(dict):
    """违规记录：{code, file, line, detail}。"""


def _rel(path: Path, root: Path) -> str:
    try:
        return path.resolve().relative_to(root.resolve()).as_posix()
    except ValueError:
        return path.as_posix()


def check_file(path: Path, root: Path) -> list[Finding]:
    findings: list[Finding] = []
    rel = _rel(path, root)
    if any(rel.startswith(p) for p in _REFLECTION_ALLOWLIST):
        allow_reflection = True
    else:
        allow_reflection = False

    try:
        tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    except SyntaxError as exc:
        return [Finding(code="SYNTAX", file=rel, line=exc.lineno or 0, detail=str(exc))]

    imports_dataclass = False
    imports_pydantic = False
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for a in node.names:
                if a.name == "dataclasses":
                    imports_dataclass = True
                if a.name.startswith("pydantic"):
                    imports_pydantic = True
                if a.name == "importlib" and not allow_reflection:
                    findings.append(Finding(
                        code=CODE_C3, file=rel, line=node.lineno,
                        detail="import importlib（反射式取用，Cython 不可静态解析）"))
        elif isinstance(node, ast.ImportFrom):
            if node.module == "dataclasses":
                imports_dataclass = True
            if (node.module or "").startswith("pydantic"):
                imports_pydantic = True

        # C1 运行时造类
        if isinstance(node, ast.Call):
            fn = node.func
            if isinstance(fn, ast.Name) and fn.id == "type" and len(node.args) >= 3:
                findings.append(Finding(
                    code=CODE_C1, file=rel, line=node.lineno,
                    detail="type(name, bases, dict) 运行时造类"))
            if isinstance(fn, ast.Name) and fn.id == "__build_class__":
                findings.append(Finding(
                    code=CODE_C1, file=rel, line=node.lineno,
                    detail="__build_class__ 运行时造类"))
            if (isinstance(fn, ast.Attribute) and fn.attr == "new_class"
                    and isinstance(fn.value, ast.Name) and fn.value.id == "types"):
                findings.append(Finding(
                    code=CODE_C1, file=rel, line=node.lineno,
                    detail="types.new_class 运行时造类"))
            # C3 getattr 变量名
            if isinstance(fn, ast.Name) and fn.id == "getattr" and len(node.args) >= 2:
                name_arg = node.args[1]
                if not isinstance(name_arg, ast.Constant) and not allow_reflection:
                    findings.append(Finding(
                        code=CODE_C3, file=rel, line=node.lineno,
                        detail="getattr(obj, <变量>) 反射式取用（仅允许字符串常量名）"))

        # C2 **kwargs
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.kwarg is not None:
            findings.append(Finding(
                code=CODE_C2, file=rel, line=node.lineno,
                detail=f"def {node.name}(**{node.kwarg.arg}) 动态关键字派发"))

    # C4 契约静态声明
    if "/contract/" in f"/{rel}" and not (imports_dataclass or imports_pydantic):
        findings.append(Finding(
            code=CODE_C4, file=rel, line=1,
            detail="契约模块未使用 dataclasses / pydantic 静态声明"))
    return findings


def scan(paths: Iterable[Path], root: Path) -> list[Finding]:
    out: list[Finding] = []
    for p in paths:
        if p.is_dir():
            out.extend(check_file(f, root) for f in sorted(p.rglob("*.py")))
        else:
            out.append(check_file(p, root))
    flat: list[Finding] = []
    for item in out:
        if isinstance(item, list):
            flat.extend(item)
        else:
            flat.append(item)
    return flat


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="内核可编译友好检查（设计 §6.1）")
    ap.add_argument("paths", nargs="+", help="待检查的文件或目录")
    ap.add_argument("--root", default=None, help="仓库根（默认取 packages 的父级）")
    ap.add_argument("--json", action="store_true", help="输出 JSON")
    args = ap.parse_args(argv)

    targets = [Path(p) for p in args.paths]
    if args.root:
        root = Path(args.root)
    else:
        root = Path.cwd()
    findings = scan(targets, root)

    if args.json:
        print(json.dumps([dict(f) for f in findings], ensure_ascii=False, indent=2))
    else:
        if not findings:
            print(f"OK: 可编译友好检查通过（{len(list(targets))} 个目标）")
        for f in findings:
            print(f"[{f['code']}] {f['file']}:{f['line']} {f['detail']}")
        print(f"\n合计违规 {len(findings)} 条")
    return 1 if findings else 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def assert_no_free_jsonb(schema_tables: dict) -> List[str]:
    """可执行判据：业务粒子表 / 边表不得有 jsonb 列。返回违规列表。

    demo 已把用户对「7 张宽表 + payload JSONB」的否决固化成这个函数。
    ⛔ 本判据必须**保持可执行**——写成文档条款就无人能拦住下一次加 jsonb。
    """
    violations: List[str] = []
    for table, cols in schema_tables.items():
        if table in _ALLOWED_JSONB_TABLES:
            continue
        for col in cols:
            name, _, typ = col.partition(":")
            if typ.strip().lower() in ("jsonb", "json"):
                violations.append(f"{table}.{name}:{typ}")
    return violations
```

```python
"""A3 判据：消费方只许 import 契约与门面，不得 import 内核内部实现。

实现方式：扫描消费方源码的 import 语句（AST），命中
`chiyu_kernel.<内部模块>` 即失败。白名单：`chiyu_kernel` / `chiyu_kernel.contract`
/ `chiyu_kernel.api`。

⛔ 为什么必须扫描源码而不是"看代码习惯"：习惯不会在 CI 里变红。
"""
from __future__ import annotations

import ast
from pathlib import Path

import pytest

ALLOWED = {"chiyu_kernel", "chiyu_kernel.contract", "chiyu_kernel.api"}


def _imports(path: Path) -> list[tuple[int, str]]:
    tree = ast.parse(path.read_text(encoding="utf-8"))
    out: list[tuple[int, str]] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            out.extend((node.lineno, a.name) for a in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            out.append((node.lineno, node.module))
    return out


def check_consumer_dir(root: Path) -> list[str]:
    bad: list[str] = []
    for p in sorted(root.rglob("*.py")):
        for lineno, mod in _imports(p):
            if mod.startswith("chiyu_kernel") and mod not in ALLOWED:
                bad.append(f"{p.relative_to(root)}:{lineno} import {mod}")
    return bad


@pytest.mark.parametrize("sample", ["violating", "clean"])
def test_guard_discriminates(tmp_path: Path, sample: str):
    """判别力自证：含违约 import 的样本必须被检出，干净样本必须通过。"""
    d = tmp_path / sample
    d.mkdir()
    if sample == "violating":
        (d / "a.py").write_text("from chiyu_kernel.store.pg_store import PgVectorStore\n")
    else:
        (d / "a.py").write_text("from chiyu_kernel.contract import TenantProfile\n")
    found = check_consumer_dir(d)
    assert bool(found) is (sample == "violating")
```

- [ ] **Step 2: 写 `vector/coverage.py`（★ 新增：覆盖率 + backfill，补 demo 缺口）**

设计稿 §2.1 要求 `ontology/` 带覆盖率与 backfill；demo 的 `ontology/backfill.py` 存在但**无监控闭环**（审计 P2-1 记为待补）。本模块把"覆盖率"做成可查询指标。

```python
"""向量覆盖率与存量回填 —— 写时物化的**监控闭环**。

⛔ 为什么覆盖率必须是可查指标而不是日志：
   "降级不中断"是好事，但降级**静默**就是坏事——真向量通道断了以后，
   系统照常运转、检索结果缓慢变差，直到有人发现"怎么搜都不准"。覆盖率
   是把这件事变成数字的**唯一**手段（demo 审计 §P2-1 记的正是这个缺口）。

口径（可执行）：
   coverage = 有真向量(provider 非 hash) 的粒子数 / 应物化粒子数
   min_coverage 由 profile 声明（默认 0.8）；低于阈值 ⇒ vector_degraded=True
   ⛔ 分母是"应物化"（类型在 profile.vectorized_types 内），不是"全部粒子"
      —— 否则不参与语义检索的粒子会稀释指标，覆盖率永远达不到阈值。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Iterable, Mapping

DEFAULT_MIN_COVERAGE = 0.8
HASH_PROVIDER = "hash"


@dataclass(frozen=True)
class CoverageReport:
    expected: int
    materialized: int
    hash_fallback: int
    min_coverage: float

    @property
    def coverage(self) -> float:
        return (self.materialized / self.expected) if self.expected else 1.0

    @property
    def degraded(self) -> bool:
        return self.coverage < self.min_coverage

    def as_metrics(self) -> Mapping[str, float]:
        return {
            "vector_coverage": round(self.coverage, 4),
            "vector_expected": float(self.expected),
            "vector_materialized": float(self.materialized),
            "vector_hash_fallback": float(self.hash_fallback),
            "vector_degraded": 1.0 if self.degraded else 0.0,
        }


def compute_coverage(rows: Iterable[Mapping], min_coverage: float = DEFAULT_MIN_COVERAGE) -> CoverageReport:
    """rows 形如 [{"provider": "model"|"model-p384"|"hash"|None}, ...]（仅应物化粒子）。"""
    expected = 0
    materialized = 0
    fallback = 0
    for r in rows:
        expected += 1
        prov = r.get("provider")
        if prov is None or prov == HASH_PROVIDER:
            fallback += 1
        else:
            materialized += 1
    return CoverageReport(expected, materialized, fallback, min_coverage)


def backfill_plan(rows: Iterable[Mapping], limit: int = 500) -> list:
    """待回填清单（有 provider=hash 或 None 的粒子 id），按 limit 截断。

    ⛔ 只产出计划不执行写入 —— 回填是**写操作**，须经决策第 0 闸（本项目纪律）。
    """
    out = []
    for r in rows:
        if r.get("provider") in (None, HASH_PROVIDER):
            out.append(r.get("particle_id"))
        if len(out) >= limit:
            break
    return [x for x in out if x]


__all__ = ["CoverageReport", "compute_coverage", "backfill_plan",
           "DEFAULT_MIN_COVERAGE", "HASH_PROVIDER"]
```

```python
"""上下文装配（L1–L4）+ KnowledgeNeeds 供需校验。

四层语义（SKILL #4）：
    L1 知识底座      静态领域知识（本体 / 受控词表 / 规格）
    L2 历史决策      先例（含**反面先例** —— 第 9 尺 fairmess 的输入）
    L3 执行协同      任务态 / 派发 / 编排上下文
    L4 治理决策      红线 / 审批 / 权限 / 合规

⛔ 供需一致性校验（为什么必须可失败）
────────────────────────────────────
第 9 尺的 relevance 尺算的是「所需维度 ∩ 已供给维度」。若供给表能静默缺少
某层（例如反例只在"检索到了"但从未"被消费"），relevance 会在**看起来正常**的
情况下恒得低分——这正是 demo 里"评分被降级为叙述留痕"的同族缺陷。
故 assemble() 返回 `supply_report`，缺口以**结构化字段**给出，调用方必须显式处理。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Iterable, Mapping, Sequence

CONTEXT_LAYERS = ("L1", "L2", "L3", "L4")
LAYER_LABELS_ZH = {
    "L1": "知识底座", "L2": "历史决策", "L3": "执行协同", "L4": "治理决策",
}


@dataclass(frozen=True)
class KnowledgeNeed:
    """需求侧编码：本决策需要哪一层、哪几个维度、哪个键。"""

    layer: str
    dims: tuple = ()
    key: str = ""

    def __post_init__(self) -> None:
        if self.layer not in CONTEXT_LAYERS:
            raise ValueError(f"layer 须 ∈ {CONTEXT_LAYERS}：{self.layer!r}")


@dataclass
class SupplyReport:
    required: tuple = ()
    supplied: tuple = ()
    missing: tuple = ()
    degraded_layers: tuple = ()

    @property
    def consistent(self) -> bool:
        return not self.missing

    def as_metrics(self) -> Mapping[str, float]:
        req = len(self.required)
        hit = req - len(self.missing)
        return {
            "context_supply_rate": round(hit / req, 4) if req else 1.0,
            "context_missing": float(len(self.missing)),
            "context_degraded_layers": float(len(self.degraded_layers)),
        }


@dataclass
class AssemblyResult:
    facts: dict = field(default_factory=dict)
    supply_report: SupplyReport = field(default_factory=SupplyReport)


def assemble(needs: Sequence[KnowledgeNeed],
             provided: Mapping[str, Iterable]) -> AssemblyResult:
    """按需装配。

    needs    KnowledgeNeed 序列（需求侧，由调用方按场景声明）
    provided {layer: [{"dim":..., "key":..., "value":...}, ...]}（供给侧）
    返回    facts + supply_report（缺口**结构化**返回，不吞不糊）
    """
    facts: dict = {}
    supplied_keys: list = []
    missing: list = []
    degraded: list = []
    for need in needs:
        rows = list(provided.get(need.layer) or [])
        if not rows:
            degraded.append(need.layer)
        hit = [r for r in rows
               if (not need.dims or _dim_of(r) in need.dims)
               and (not need.key or r.get("key") == need.key)]
        if hit:
            supplied_keys.append((need.layer, need.key))
            for r in hit:
                facts.setdefault(need.layer, []).append(r.get("value"))
        else:
            missing.append((need.layer, need.key))
    return AssemblyResult(
        facts=facts,
        supply_report=SupplyReport(
            required=tuple((n.layer, n.key) for n in needs),
            supplied=tuple(supplied_keys),
            missing=tuple(missing),
            degraded_layers=tuple(degraded),
        ),
    )


def _dim_of(row: Mapping) -> str:
    d = row.get("dim")
    if isinstance(d, dict):
        return d.get("key") or d.get("dim") or ""
    return d or ""


__all__ = ["CONTEXT_LAYERS", "LAYER_LABELS_ZH", "KnowledgeNeed",
           "SupplyReport", "AssemblyResult", "assemble"]
```

- [ ] **Step 5: 写 `actions/registry.py`（★ 新建：demo 未收敛）**

审计判定：demo 有 **59 个端点**、未按 R1–R6 审计收敛、R6 force 仅 2 处。本模块提供收敛的**机制**（不含具体 Action 清单——那是 L4）。

```python
"""Action 表面注册表（R1–R6 基数规则的**可执行形式**）。

⛔ 为什么要有注册表而不是继续加端点
──────────────────────────────────
demo 的 59 个端点里出现"同一能力多端点并存"（如 `/api/retro/*` 4 个），
这是 R5（一能力一 Action）被违反的典型。加端点不会报错，所以没人拦得住 ——
除非有一个注册表在**注册期**就拒绝重复。

R1–R6（判定规则）：
    R1 一个业务域 5–9 个 Action（过多=CRUD 爆炸，过少=能力不足）
    R2 Action 名 = 动词 + 对象（`submit_request` 而非 `requestSubmit`）
    R3 读操作不单独占位（读并入对象的 query Action）
    R4 写操作必须有决策锚（decision_id）
    R5 一个能力一个 Action（同能力的多入口必须折叠）
    R6 高危写必须有 force 确认位
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple

ACTION_MIN, ACTION_MAX = 5, 9
_NAME_RE = re.compile(r"^[a-z][a-z0-9]*(_[a-z0-9]+)+$")
_HIGH_RISK_HINTS = ("delete", "purge", "reset", "drop", "approve", "release")


@dataclass(frozen=True)
class ActionSpec:
    name: str
    capability: str                 # R5：同一 capability 只能有一个 Action
    mutating: bool = False          # R4：写操作必须有决策锚
    high_risk: bool = False         # R6：高危写必须有 force
    has_force: bool = False

    def violations(self) -> List[str]:
        out: List[str] = []
        if not _NAME_RE.match(self.name):
            out.append(f"R2 命名不合规（须 verb_object）：{self.name}")
        if self.high_risk and not self.has_force:
            out.append(f"R6 高危写缺 force 确认位：{self.name}")
        if not self.high_risk and self.has_force:
            out.append(f"R6 非高危动作不应带 force 位：{self.name}")
        return out


class ActionRegistry:
    def __init__(self) -> None:
        self._by_name: Dict[str, ActionSpec] = {}
        self._by_capability: Dict[str, str] = {}

    def register(self, spec: ActionSpec) -> None:
        if spec.name in self._by_name:
            raise ValueError(f"重复注册 Action：{spec.name}")
        owner = self._by_capability.get(spec.capability)
        if owner is not None:
            raise ValueError(
                f"R5 违反：capability {spec.capability!r} 已由 {owner!r} 占用，"
                f"新增 {spec.name!r} ⇒ 应折叠到既有 Action，而不是新开端点")
        self._by_name[spec.name] = spec
        self._by_capability[spec.capability] = spec.name

    def audit(self) -> Tuple[List[str], List[str]]:
        """返回 (errors, warnings)。errors 非空 ⇒ 表面不合规，不得发布。"""
        errors: List[str] = []
        warnings: List[str] = []
        n = len(self._by_name)
        if n < ACTION_MIN:
            warnings.append(f"R1 Action 数偏少（{n} < {ACTION_MIN}）")
        if n > ACTION_MAX:
            errors.append(f"R1 Action 数超限（{n} > {ACTION_MAX}）⇒ CRUD 爆炸")
        for spec in self._by_name.values():
            errors.extend(spec.violations())
        return errors, warnings

    def by_name(self, name: str) -> Optional[ActionSpec]:
```

```python
"""事件总线（SKILL #8 的机制层）。

⛔ 与 `event` 表的关系
──────────────────────
`bj.event` 是**持久化事实源**（append-only）；本模块是**进程内分发**。
    bus.publish() → ① 落 event 表（持久，不可丢）→ ② 通知订阅者（可丢，可重放）
⛔ 顺序不可颠倒：先落库后分发。反了会出现"订阅者处理了但事件没留下"，
   回放时状态对不上（本项目已撞过同族缺陷：中间态静默）。

⛔ 订阅者异常不得中断发布（否则一个坏的订阅者让全系统的事件都发不出去），
   但必须**计数并留痕** —— 静默吞异常会让"事件发了但没人处理"永远不被发现。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Callable, Dict, List, Mapping, Tuple

from chiyu_kernel.contract import PROBE_ASSERTIONS  # noqa: F401  （契约可达性自证）

Subscriber = Callable[[Mapping], None]


@dataclass
class PublishReport:
    event_id: str
    delivered: int = 0
    failed: Tuple[str, ...] = ()

    @property
    def healthy(self) -> bool:
        return not self.failed


class EventBus:
    def __init__(self) -> None:
        self._subs: Dict[str, List[Tuple[str, Subscriber]]] = {}

    def subscribe(self, event_type: str, name: str, fn: Subscriber) -> None:
        if not callable(fn):
            raise TypeError(f"订阅者必须可调用：{name}")
        self._subs.setdefault(event_type, []).append((name, fn))

    def subscribers(self, event_type: str) -> Tuple[str, ...]:
        return tuple(n for n, _ in self._subs.get(event_type, []))

    def publish(self, event: Mapping, persist: Callable[[Mapping], None]) -> PublishReport:
        """persist 先执行（持久化），再分发。"""
        eid = str(event.get("id") or "")
        if not eid:
            raise ValueError("事件必须有 id（否则无法回放与去重）")
        persist(event)                                  # ① 先落库
        delivered = 0
        failed: List[str] = []
        for name, fn in self._subs.get(str(event.get("event_type")), []):
            try:
                fn(event)                               # ② 再分发
                delivered += 1
            except Exception:                           # noqa: BLE001
                failed.append(name)                     # 计数留痕，不静默
        return PublishReport(event_id=eid, delivered=delivered, failed=tuple(failed))


__all__ = ["EventBus", "PublishReport", "Subscriber"]
```

```python
"""SHA-256 链式审计（SKILL #10 的机制层）。

⛔ 为什么是链式而不是逐条哈希
────────────────────────────
逐条哈希只能证明"这条没被改"；链式还能证明"**没有中间记录被删掉**"——
后者才是审计被绕过的常见形式（删掉不利记录，其余保持自洽）。

    h_i = sha256( h_{i-1} + canonical(record_i) )，h_0 = "0"*64

canonical 规则：按 AUDIT_REQUIRED_FIELDS 顺序取字段，缺失即 "∅"（占位符而非跳过
—— 跳过会让"删字段"与"原值就是空"产生同一哈希，蓄意者可借此抹除字段）。
"""
from __future__ import annotations

import hashlib
import json
from dataclasses import dataclass
from typing import Iterable, List, Mapping, Sequence

from chiyu_kernel.contract import (
    AUDIT_GENESIS_HASH, AUDIT_HASH_ALGO, AUDIT_REQUIRED_FIELDS,
)

EMPTY_PLACEHOLDER = "∅"


def canonical(record: Mapping, fields: Sequence[str] = AUDIT_REQUIRED_FIELDS) -> str:
    payload = {f: record.get(f, EMPTY_PLACEHOLDER) for f in fields}
    return json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def chain_hash(prev_hash: str, record: Mapping) -> str:
    if AUDIT_HASH_ALGO != "sha256":
        raise ValueError(f"未支持的审计哈希算法：{AUDIT_HASH_ALGO}")
    h = hashlib.sha256()
    h.update(prev_hash.encode("utf-8"))
    h.update(canonical(record).encode("utf-8"))
    return h.hexdigest()


def build_chain(records: Iterable[Mapping]) -> List[str]:
    prev = AUDIT_GENESIS_HASH
    out: List[str] = []
    for r in records:
        prev = chain_hash(prev, r)
        out.append(prev)
    return out


@dataclass(frozen=True)
class VerifyResult:
    ok: bool
    checked: int
    first_bad_index: int = -1
    reason: str = ""


def verify_chain(records: Sequence[Mapping], expected: Sequence[str]) -> VerifyResult:
    """重算并比对。返回首个失配位置（-1 表示全链通过）。"""
    if len(records) != len(expected):
        return VerifyResult(False, min(len(records), len(expected)), -1,
                            f"长度不一致：records={len(records)} expected={len(expected)}")
    prev = AUDIT_GENESIS_HASH
    for i, r in enumerate(records):
        prev = chain_hash(prev, r)
        if prev != expected[i]:
            return VerifyResult(False, i + 1, i, f"第 {i} 条哈希失配")
    return VerifyResult(True, len(records))


__all__ = ["canonical", "chain_hash", "build_chain", "verify_chain",
           "VerifyResult", "EMPTY_PLACEHOLDER"]
```
